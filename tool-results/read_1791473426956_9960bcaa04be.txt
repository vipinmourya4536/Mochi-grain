    13→   History Trends – large interactive moisture chart

    23→type RangeId = Exclude<TrendsRange, 'auto'>;
    24→type MetricId = TrendsMetric;
    25→
    26→const RANGES: { id: RangeId; label: string; ms: number | null }[] = [
    27→  { id: '24h', label: '24H', ms: 24 * 3600e3 },
    28→  { id: '7d', label: '7D', ms: 7 * 24 * 3600e3 },
    29→  { id: '30d', label: '30D', ms: 30 * 24 * 3600e3 },
    30→  { id: 'all', label: 'ALL', ms: null },
    31→];
    32→
    33→const SCRUB_AUTO_CLEAR_MS = 3200;
    34→const TOOLTIP_CLAMP = 50;
    35→
    36→const W = 358;
    37→const H = 216;
    38→const PAD_X = 18;
    39→const PAD_TOP = 18;
    40→const PAD_BOT = 24;
    41→
    42→interface TrendPt {
    43→  x: number;
    44→  y: number;
    45→  v: number;
    46→  t: number;
    47→  state: string;
    48→}
    49→
    50→interface TrendDir {
    51→  dir: 'up' | 'down' | 'flat';
    52→  delta: number;
    53→  labelKey: string;
    54→}
    55→
    56→/** Direction + signed delta across the visible range (per metric) */
    57→function trendDirection(delta: number, metric: MetricId): TrendDir {
    58→  const eps = metric === 'moisture' ? 0.15 : 0.3;
    59→  if (delta > eps) return { dir: 'up', delta, labelKey: 'history.trend_rising' };
    60→  if (delta < -eps) return { dir: 'down', delta, labelKey: 'history.trend_falling' };
    61→  return { dir: 'flat', delta, labelKey: 'history.trend_stable' };
    62→}
    63→
    64→function filterEntries(entries: HistoryEntry[], range: RangeId): HistoryEntry[] {
    65→  const r = RANGES.find((x) => x.id === range);
    66→  if (!r || r.ms == null) return entries;
    67→  const cutoff = Date.now() - r.ms;
    68→  return entries.filter((e) => e.reading.timestamp >= cutoff);
    69→}
    70→
    71→/** Smallest range with ≥2 points, falling back to ALL */
    72→function detectRange(entries: HistoryEntry[]): RangeId {
    73→  for (const r of RANGES) {
    74→    if (filterEntries(entries, r.id).length >= 2) return r.id;
    75→  }
    76→  return 'all';
    77→}

    79→/** Monotone cubic Bézier (Fritsch–Carlson) — shared curve algorithm */
    80→function monotoneCubic(pts: { x: number; y: number }[]) {
    81→  const n = pts.length;
    82→  if (n < 2) return [];
    83→  const dx: number[] = [];
    84→  const dy: number[] = [];
    85→  const m: number[] = [];
    86→  for (let i = 0; i < n - 1; i++) {
    87→    dx.push(pts[i + 1].x - pts[i].x);
    88→    dy.push(pts[i + 1].y - pts[i].y);
    89→    m.push(dy[i] / (dx[i] || 1));
    90→  }
    91→  const c: number[] = [m[0]];
    92→  for (let i = 1; i < n - 1; i++) {
    93→    if (m[i - 1] * m[i] <= 0) c.push(0);
    94→    else {
    95→      const common = dx[i - 1] + dx[i];
    96→      c.push((3 * common) / ((common + dx[i]) / m[i - 1] + (common + dx[i - 1]) / m[i]));
    97→    }
    98→  }
    99→  c.push(m[n - 2]);
   100→  for (let i = 0; i < n - 1; i++) {
   101→    const acc = c[i] / (m[i] || 1);
   102→    const succ = c[i + 1] / (m[i] || 1);
   103→    if (acc < 0 || succ < 0) {
   104→      c[i] = 0;
   105→      c[i + 1] = 0;
   106→    }
   107→  }
   108→  const segs: { x: number; y: number; cp1x: number; cp1y: number; cp2x: number; cp2y: number }[] = [];
   109→  for (let i = 0; i < n - 1; i++) {
   110→    const p0 = pts[i];
   111→    const p1 = pts[i + 1];
   112→    const segDx = dx[i];
   113→    segs.push({
   114→      x: p1.x, y: p1.y,
   115→      cp1x: p0.x + segDx / 3, cp1y: p0.y + (segDx / 3) * c[i],
   116→      cp2x: p1.x - segDx / 3, cp2y: p1.y - (segDx / 3) * c[i + 1],
   117→    });
   118→  }
   119→  return segs;
   120→}
   121→
   122→export function HistoryTrends({ entries }: { entries: HistoryEntry[] }) {
   123→  const { settings, updateSettings } = useGrainStore();
   124→  const lang = settings.language as AppLanguage;
   125→  const accentHex = getAccentHex(settings.accentColor);
   126→  const { safe, warn, critical } = settings.thresholds;
   127→
   128→  // Metric + range persist across sessions via AppSettings ('auto' re-detects)
   129→  const [rangeSel, setRangeSel] = useState<RangeId>(() =>
   130→    settings.trendsRange === 'auto' ? detectRange(entries) : settings.trendsRange,
   131→  );
   132→  const [metric, setMetric] = useState<MetricId>(settings.trendsMetric);
   133→
   134→  const filtered = useMemo(() => filterEntries(entries, rangeSel), [entries, rangeSel]);
   135→
   136→  /* ── Geometry ── */
   137→  const geo = useMemo(() => {
   138→    if (filtered.length < 2) return null;
   139→
   140→    // Ascending by time
   141→    const sorted = [...filtered].sort((a, b) => a.reading.timestamp - b.reading.timestamp);
   142→    const getV = (e: HistoryEntry) =>
   143→      metric === 'moisture' ? e.reading.moisture : e.reading.temperature;
   144→    // Temperature is required for temp mode — tolerate legacy entries without it
   145→    const usable = metric === 'temp'
   146→      ? sorted.filter((e) => typeof e.reading.temperature === 'number')
   147→      : sorted;
   148→    if (usable.length < 2) return null;
   149→
   150→    const values = usable.map(getV);
   151→
   152→    const tMin = usable[0].reading.timestamp;
   153→    const tMax = usable[usable.length - 1].reading.timestamp;
   154→    const tRange = tMax - tMin || 1;
   155→
   156→    let lo: number;
   157→    let hi: number;
   158→    let vPad: number;
   159→    if (metric === 'moisture') {
   160→      // Include thresholds in the domain so zone bands are always visible
   161→      lo = Math.min(...values, safe);
   162→      hi = Math.max(...values, critical);
   163→      vPad = Math.max((hi - lo) * 0.25, 1.2);
   164→    } else {
   165→      lo = Math.min(...values);
   166→      hi = Math.max(...values);
   167→      vPad = Math.max((hi - lo) * 0.3, 1.5);
   168→    }
   169→    const vLo = lo - vPad;
   170→    const vHi = hi + vPad;
   171→    const vSpan = vHi - vLo;
   172→
   173→    const drawW = W - PAD_X * 2;
   174→    const drawH = H - PAD_TOP - PAD_BOT;
   175→
   176→    const xOf = (ts: number) => PAD_X + ((ts - tMin) / tRange) * drawW;
   177→    const yOf = (v: number) => PAD_TOP + drawH - ((v - vLo) / vSpan) * drawH;
   178→
   179→    const pts: TrendPt[] = usable.map((e) => ({
   180→      x: xOf(e.reading.timestamp),
   181→      y: yOf(getV(e)),
   182→      v: getV(e),
   183→      t: e.reading.timestamp,
   184→      state: e.decision.state,
   185→    }));
   186→
   187→    // Curve path
   188→    let pathD = `M${pts[0].x.toFixed(2)},${pts[0].y.toFixed(2)}`;
   189→    for (const seg of monotoneCubic(pts)) {
   190→      pathD += ` C${seg.cp1x.toFixed(2)},${seg.cp1y.toFixed(2)} ${seg.cp2x.toFixed(2)},${seg.cp2y.toFixed(2)} ${seg.x.toFixed(2)},${seg.y.toFixed(2)}`;
   191→    }
   192→    const fillD = `${pathD} L${pts[pts.length - 1].x.toFixed(2)},${H - PAD_BOT} L${pts[0].x.toFixed(2)},${H - PAD_BOT} Z`;
   193→
   194→    // Zone bands (clamped to the plot area) — moisture only (thresholds are moisture-specific)
   195→    const plotBottom = H - PAD_BOT;
   196→    const band = (vFrom: number, vTo: number) => {
   197→      const yA = Math.max(PAD_TOP, Math.min(plotBottom, yOf(vFrom)));
   198→      const yB = Math.max(PAD_TOP, Math.min(plotBottom, yOf(vTo)));
   199→      return { y: Math.min(yA, yB), h: Math.abs(yB - yA) };
   200→    };
   201→    const critBand = band(Math.max(critical, vLo), vHi);        // above critical
   202→    const warnBand = band(safe, critical);                       // between safe and critical
   203→    const safeBand = band(vLo, safe);                            // below safe
   204→
   205→    // Stats
   206→    const avg = values.reduce((s, v) => s + v, 0) / values.length;
   207→    const min = Math.min(...values);
   208→    const max = Math.max(...values);
   209→
   210→    // Dot thinning for long ranges
   211→    const dotStep = pts.length > 60 ? Math.ceil(pts.length / 48) : 1;
   212→
   213→    // Time axis labels: first / mid / last
   214→    const midIdx = Math.floor(pts.length / 2);
   215→    const fmt = (ts: number) => (rangeSel === '24h' ? format(ts, 'HH:mm') : format(ts, 'MMM d'));
   216→    const timeLabels = [
   217→      { x: pts[0].x, anchor: 'start' as const, label: fmt(tMin) },
   218→      ...(pts.length > 6 ? [{ x: pts[midIdx].x, anchor: 'middle' as const, label: fmt(usable[midIdx].reading.timestamp) }] : []),
   219→      { x: pts[pts.length - 1].x, anchor: 'end' as const, label: fmt(tMax) },
   220→    ];
   221→
   222→    return {
   223→      pts, pathD, fillD, critBand, warnBand, safeBand, avg, min, max,
   224→      avgY: yOf(avg), dotStep, timeLabels,
   225→      safeY: yOf(safe), criticalY: yOf(critical),
   226→      trend: trendDirection(values[values.length - 1] - values[0], metric),
   227→    };
   228→  }, [filtered, safe, critical, rangeSel, metric]);
   229→
   230→  /* ── Scrub state (timestamp-anchored, same pattern as hero sparkline) ── */
   231→  const [scrubSel, setScrubSel] = useState<{ t: number; leftPx: number; topPx: number } | null>(null);
   232→  const svgRef = useRef<SVGSVGElement | null>(null);
   233→  const wrapRef = useRef<HTMLDivElement | null>(null);
   234→  const draggingRef = useRef(false);
   235→  const autoClearRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   236→
   237→  const scrubPt = useMemo(
   238→    () => (scrubSel && geo ? geo.pts.find((p) => p.t === scrubSel.t) ?? null : null),
   239→    [geo, scrubSel],
   240→  );
   241→  const scrubIdx = useMemo(
   242→    () => (geo && scrubSel ? geo.pts.findIndex((p) => p.t === scrubSel.t) : -1),
   243→    [geo, scrubSel],
   244→  );
   245→
   246→  const clearAutoTimer = useCallback(() => {
   247→    if (autoClearRef.current) {
   248→      clearTimeout(autoClearRef.current);
   249→      autoClearRef.current = null;
   250→    }
   251→  }, []);
   252→
   253→  const clearScrub = useCallback(() => {
   254→    clearAutoTimer();
   255→    draggingRef.current = false;
   256→    setScrubSel(null);
   257→  }, [clearAutoTimer]);
   258→
   259→  useEffect(() => () => clearAutoTimer(), [clearAutoTimer]);
   260→
   261→  /** Switch range or metric: clears any active scrub (selection is scope-scoped)
   262→   *  and persists the choice so the chart reopens in the same configuration. */
   263→  const selectRange = useCallback(
   264→    (id: RangeId) => {
   265→      clearAutoTimer();
   266→      draggingRef.current = false;
   267→      setScrubSel(null);
   268→      setRangeSel(id);
   269→      updateSettings({ trendsRange: id });
   270→    },
   271→    [clearAutoTimer, updateSettings],
   272→  );
   273→
   274→  const selectMetric = useCallback(
   275→    (m: MetricId) => {
   276→      clearAutoTimer();
   277→      draggingRef.current = false;
   278→      setScrubSel(null);
   279→      setMetric(m);
   280→      updateSettings({ trendsMetric: m });
   281→    },
   282→    [clearAutoTimer, updateSettings],
   283→  );
   284→
   285→  const anchorFor = useCallback(
   286→    (i: number): { leftPx: number; topPx: number } | null => {
   287→      const svg = svgRef.current;
   288→      const wrap = wrapRef.current;
   289→      if (!svg || !wrap || !geo || i < 0 || i >= geo.pts.length) return null;
   290→      const ctm = svg.getScreenCTM();
   291→      if (!ctm) return null;
   292→      const pt = geo.pts[i];
   293→      let cx: number;
   294→      let cy: number;
   295→      if (typeof DOMPoint === 'function') {
   296→        const sp = new DOMPoint(pt.x, pt.y).matrixTransform(ctm);
   297→        cx = sp.x; cy = sp.y;
   298→      } else {
   299→        cx = pt.x * ctm.a + pt.y * ctm.c + ctm.e;
   300→        cy = pt.x * ctm.b + pt.y * ctm.d + ctm.f;
   301→      }
   302→      const rect = wrap.getBoundingClientRect();
   303→      const leftPx = Math.min(Math.max(cx - rect.left, TOOLTIP_CLAMP), Math.max(rect.width - TOOLTIP_CLAMP, TOOLTIP_CLAMP));
   304→      const topPx = cy - rect.top;
   305→      return { leftPx, topPx };
   306→    },
   307→    [geo],
   308→  );
   309→
   310→  const nearestIndex = useCallback(
   311→    (vx: number): number => {
   312→      if (!geo || geo.pts.length === 0) return -1;
   313→      const pts = geo.pts;
   314→      let lo = 0;
   315→      let hi = pts.length - 1;
   316→      while (hi - lo > 1) {
   317→        const mid = (lo + hi) >> 1;
   318→        if (pts[mid].x < vx) lo = mid;
   319→        else hi = mid;
   320→      }
   321→      return Math.abs(pts[lo].x - vx) <= Math.abs(pts[hi].x - vx) ? lo : hi;
   322→    },
   323→    [geo],
   324→  );
   325→
   326→  const scrubTo = useCallback(
   327→    (i: number) => {
   328→      const anchor = anchorFor(i);
   329→      if (!anchor || !geo) return;
   330→      const sel = { t: geo.pts[i].t, ...anchor };
   331→      setScrubSel((prev) => {
   332→        if (prev && prev.t !== sel.t) haptic(settings.hapticsEnabled, 4);
   333→        return sel;
   334→      });
   335→    },
   336→    [anchorFor, geo, settings.hapticsEnabled],
   337→  );
   338→
   339→  const clientToViewBox = useCallback((clientX: number, clientY: number) => {
   340→    const svg = svgRef.current;
   341→    if (!svg) return null;
   342→    const ctm = svg.getScreenCTM();
   343→    if (!ctm) return null;
   344→    if (typeof DOMPoint === 'function') {
   345→      const p = new DOMPoint(clientX, clientY).matrixTransform(ctm.inverse());
   346→      return { x: p.x, y: p.y };
   347→    }
   348→    const inv = ctm.inverse();
   349→    return {
   350→      x: clientX * inv.a + clientY * inv.c + inv.e,
   351→      y: clientX * inv.b + clientY * inv.d + inv.f,
   352→    };
   353→  }, []);
   354→
   355→  const onPointerDown = useCallback(
   356→    (e: React.PointerEvent<SVGSVGElement>) => {
   357→      if (!geo || geo.pts.length < 2) return;
   358→      clearAutoTimer();
   359→      draggingRef.current = true;
   360→      try { svgRef.current?.setPointerCapture(e.pointerId); } catch { /* no-op */ }
   361→      const p = clientToViewBox(e.clientX, e.clientY);
   362→      if (p) scrubTo(nearestIndex(p.x));
   363→    },
   364→    [geo, clearAutoTimer, clientToViewBox, nearestIndex, scrubTo],
   365→  );
   366→
   367→  const onPointerMove = useCallback(
   368→    (e: React.PointerEvent<SVGSVGElement>) => {
   369→      if (!geo || geo.pts.length < 2) return;
   370→      const hoverScrub = e.pointerType === 'mouse' && e.buttons === 0;
   371→      if (!draggingRef.current && !hoverScrub) return;
   372→      const p = clientToViewBox(e.clientX, e.clientY);
   373→      if (p) scrubTo(nearestIndex(p.x));
   374→    },
   375→    [geo, clientToViewBox, nearestIndex, scrubTo],
   376→  );
   377→
   378→  const onPointerUp = useCallback(
   379→    (e: React.PointerEvent<SVGSVGElement>) => {
   380→      draggingRef.current = false;
   381→      try { svgRef.current?.releasePointerCapture(e.pointerId); } catch { /* no-op */ }
   382→      if (e.pointerType !== 'mouse') {
   383→        clearAutoTimer();
   384→        autoClearRef.current = setTimeout(() => setScrubSel(null), SCRUB_AUTO_CLEAR_MS);
   385→      }
   386→    },
   387→    [clearAutoTimer],
   388→  );
   389→
   390→  const onPointerLeave = useCallback(
   391→    (e: React.PointerEvent<SVGSVGElement>) => {
   392→      if (e.pointerType === 'mouse' && !draggingRef.current) setScrubSel(null);
   393→    },
   394→    [],
   395→  );
   396→
   397→  const onKeyDown = useCallback(
   398→    (e: React.KeyboardEvent<HTMLDivElement>) => {
   399→      if (!geo || geo.pts.length < 2) return;
   400→      const lastIdx = geo.pts.length - 1;
   401→      if (e.key === 'Escape') { clearScrub(); return; }
   402→      let next: number | null = null;
   403→      const cur = scrubIdx >= 0 ? scrubIdx : null;
   404→      if (e.key === 'ArrowLeft') next = Math.max(0, (cur ?? lastIdx) - 1);
   405→      else if (e.key === 'ArrowRight') next = Math.min(lastIdx, (cur ?? lastIdx - 1) + 1);
   406→      else if (e.key === 'Home') next = 0;
   407→      else if (e.key === 'End') next = lastIdx;
   408→      if (next === null) return;
   409→      e.preventDefault();
   410→      clearAutoTimer();
   411→      scrubTo(next);
   412→    },
   413→    [geo, scrubIdx, clearScrub, clearAutoTimer, scrubTo],
   414→  );
   415→
   416→  const onBlur = useCallback(() => clearScrub(), [clearScrub]);
   417→
   418→  /* ── Render ── */
   419→  const trend = geo?.trend ?? null;
   420→  const unitSuffix = metric === 'moisture' ? '%' : '°C';
   421→  const isMoisture = metric === 'moisture';
   422→
   423→  return (
   424→    <div className="grain-trends-card mb-4" role="group" aria-label={t('history.trends_title', lang)}>
   425→      {/* Header row 1: title + trend direction chip */}
   426→      <div className="flex items-center gap-2 mb-2">
   427→        <ChartLine size={13} weight="bold" style={{ color: 'var(--gm-accent)' }} />
   428→        <span className="text-[9px] font-bold tracking-[0.18em] uppercase" style={{ color: 'var(--gm-text-tertiary)' }}>
   429→          {t('history.trends_title', lang)}
   430→        </span>
   431→        {trend && geo && (
   432→          <span
   433→            className="grain-trend-chip grain-tabular ml-auto"
   434→            style={{
   435→              color: trend.dir === 'flat'
   436→                ? 'var(--gm-text-secondary)'
   437→                : isMoisture
   438→                  ? (trend.dir === 'up' ? 'var(--gm-up)' : 'var(--gm-down)')
   439→                  : 'var(--gm-accent)',
   440→            }}
   441→            role="status"
   442→            aria-label={`${t(trend.labelKey, lang)} ${Math.abs(trend.delta).toFixed(1)}${unitSuffix}`}
   443→          >
   444→            {trend.dir === 'up'
   445→              ? <TrendUp size={11} weight="bold" />
   446→              : trend.dir === 'down'
   447→                ? <TrendDown size={11} weight="bold" />
   448→                : <Minus size={11} weight="bold" />}
   449→            {(trend.delta > 0 ? '+' : '') + trend.delta.toFixed(1)}{unitSuffix}
   450→          </span>
   451→        )}
   452→      </div>
   453→
   454→      {/* Header row 2: metric toggle + range selector */}
   455→      <div className="flex items-center justify-between mb-3">
   456→        <div className="grain-seg grain-seg-metric" role="tablist" aria-label={t('history.trends_title', lang)}>
   457→          <button
   458→            role="tab"
   459→            aria-selected={metric === 'moisture'}
   460→            className={metric === 'moisture' ? 'active' : ''}
   461→            onClick={() => { selectMetric('moisture'); haptic(settings.hapticsEnabled, 6); }}
   462→          >
   463→            {t('hero.moisture', lang)}
   464→          </button>
   465→          <button
   466→            role="tab"
   467→            aria-selected={metric === 'temp'}
   468→            className={metric === 'temp' ? 'active' : ''}
   469→            onClick={() => { selectMetric('temp'); haptic(settings.hapticsEnabled, 6); }}
   470→          >
   471→            {t('metric.temp', lang)}
   472→          </button>
   473→        </div>
   474→        <div className="grain-seg" role="tablist" aria-label={t('history.trends_title', lang)}>
   475→          {RANGES.map((r) => (
   476→            <button
   477→              key={r.id}
   478→              role="tab"
   479→              aria-selected={rangeSel === r.id}
   480→              className={rangeSel === r.id ? 'active' : ''}
   481→              onClick={() => { selectRange(r.id); haptic(settings.hapticsEnabled, 6); }}
   482→            >
   483→              {r.label}
   484→            </button>
   485→          ))}
   486→        </div>
   487→      </div>
   488→
   489→      {/* Chart / states */}
   490→      {!geo ? (
   491→        <p className="text-[11px] leading-relaxed py-6 text-center" style={{ color: 'var(--gm-text-tertiary)' }}>
   492→          {filtered.length === 0
   493→            ? t('history.trends_empty', lang)
   494→            : t('history.trends_need_more', lang)}
   495→        </p>
   496→      ) : (
   497→        <>
   498→          <div
   499→            ref={wrapRef}
   500→            className="relative w-full grain-trends-wrap"
   501→            tabIndex={0}
   502→            role="application"
   503→            aria-label="Moisture trend chart. Use left and right arrow keys to inspect readings."
   504→            onKeyDown={onKeyDown}
   505→            onBlur={onBlur}
   506→          >
   507→            <svg
   508→              ref={svgRef}
   509→              viewBox={`0 0 ${W} ${H}`}
   510→              className="w-full h-auto grain-trends-svg"
   511→              preserveAspectRatio="xMidYMid meet"
   512→              onPointerDown={onPointerDown}
   513→              onPointerMove={onPointerMove}
   514→              onPointerUp={onPointerUp}
   515→              onPointerCancel={onPointerUp}
   516→              onPointerLeave={onPointerLeave}
   517→            >
   518→              <defs>
   519→                <linearGradient id="trendFill" x1="0" y1="0" x2="0" y2="1">
   520→                  <stop offset="0%" style={{ stopColor: 'var(--gm-accent)' }} stopOpacity="0.26" />
   521→                  <stop offset="70%" style={{ stopColor: 'var(--gm-accent)' }} stopOpacity="0.05" />
   522→                  <stop offset="100%" style={{ stopColor: 'var(--gm-accent)' }} stopOpacity="0" />
   523→                </linearGradient>
   524→              </defs>
   525→
   526→              {/* Horizontal grid hairlines – subtle instrument feel */}
   527→              {[0.25, 0.5, 0.75].map((f) => {
   528→                const gy = PAD_TOP + (H - PAD_TOP - PAD_BOT) * f;
   529→                return (
   530→                  <line
   531→                    key={f}
   532→                    x1={PAD_X} y1={gy} x2={W - PAD_X} y2={gy}
   533→                    stroke="var(--gm-separator)" strokeWidth="0.6"
   534→                  />
   535→                );
   536→              })}
   537→
   538→              {/* Threshold zone bands — moisture only (thresholds are moisture-specific) */}
   539→              {isMoisture && (
   540→                <>
   541→                  <rect x={PAD_X} y={geo.critBand.y} width={W - PAD_X * 2} height={geo.critBand.h} fill="rgba(239, 68, 68, 0.10)" rx="4" />
   542→                  <rect x={PAD_X} y={geo.warnBand.y} width={W - PAD_X * 2} height={geo.warnBand.h} fill="rgba(245, 158, 11, 0.09)" rx="4" />
   543→                  <rect x={PAD_X} y={geo.safeBand.y} width={W - PAD_X * 2} height={geo.safeBand.h} fill="rgba(34, 197, 94, 0.09)" rx="4" />
   544→                </>
   545→              )}
   546→
   547→              {/* Threshold hairlines + labels — moisture only */}
   548→              {isMoisture && (
   549→                <>
   550→                  <line x1={PAD_X} y1={geo.safeY} x2={W - PAD_X} y2={geo.safeY} stroke="rgba(34,197,94,0.35)" strokeWidth="0.75" strokeDasharray="3,4" />
   551→                  <line x1={PAD_X} y1={geo.criticalY} x2={W - PAD_X} y2={geo.criticalY} stroke="rgba(239,68,68,0.35)" strokeWidth="0.75" strokeDasharray="3,4" />
   552→                  <text
   553→                    x={W - PAD_X - 2} y={geo.safeY - 3} textAnchor="end"
   554→                    className="grain-trend-thresh grain-trend-thresh-safe"
   555→                    fontSize="7" fontWeight="700" fontFamily="system-ui, sans-serif"
   556→                  >
   557→                    {safe}%
   558→                  </text>
   559→                  <text
   560→                    x={W - PAD_X - 2} y={geo.criticalY - 3} textAnchor="end"
   561→                    className="grain-trend-thresh grain-trend-thresh-crit"
   562→                    fontSize="7" fontWeight="700" fontFamily="system-ui, sans-serif"
   563→                  >
   564→                    {critical}%
   565→                  </text>
   566→                </>
   567→              )}
   568→
   569→              {/* Area + curve */}
   570→              <path d={geo.fillD} fill="url(#trendFill)" stroke="none" />
   571→              <path
   572→                d={geo.pathD}
   573→                fill="none"
   574→                stroke={accentHex}
   575→                strokeWidth="2"
   576→                strokeLinecap="round"
   577→                strokeLinejoin="round"
   578→              />
   579→
   580→              {/* Avg reference line */}
   581→              <line
   582→                x1={PAD_X} y1={geo.avgY} x2={W - PAD_X} y2={geo.avgY}
   583→                stroke={accentHex} strokeOpacity="0.4" strokeWidth="0.9" strokeDasharray="5,4"
   584→                opacity={scrubPt ? 0.35 : 1}
   585→              />
   586→
   587→              {/* Risk-colored dots (moisture) / accent dots (temp) */}
   588→              {geo.pts.map((p, i) =>
   589→                (i % geo.dotStep === 0 || i === geo.pts.length - 1) ? (
   590→                  <circle
   591→                    key={p.t}
   592→                    cx={p.x} cy={p.y}
   593→                    r={i === geo.pts.length - 1 ? 3.2 : 2.1}
   594→                    fill={isMoisture ? getRiskColor(p.state, settings.accentColor) : accentHex}
   595→                    stroke="rgba(0,0,0,0.18)"
   596→                    strokeWidth="0.6"
   597→                    opacity={scrubPt && scrubPt.t !== p.t ? 0.55 : 1}
   598→                  />
   599→                ) : null,
   600→              )}
   601→
   602→              {/* Scrub guide + marker */}
   603→              {scrubPt && (
   604→                <g>
   605→                  <line
   606→                    x1={scrubPt.x} y1={PAD_TOP - 4}
   607→                    x2={scrubPt.x} y2={H - PAD_BOT + 4}
   608→                    stroke="var(--gm-text-tertiary)"
   609→                    strokeOpacity="0.55"
   610→                    strokeWidth="1"
   611→                    strokeDasharray="2,3"
   612→                  />
   613→                  <circle cx={scrubPt.x} cy={scrubPt.y} r={7} fill="rgba(255,255,255,0.14)" />
   614→                  <circle
   615→                    cx={scrubPt.x} cy={scrubPt.y} r={4}
   616→                    fill={isMoisture ? getRiskColor(scrubPt.state, settings.accentColor) : accentHex}
   617→                    stroke="#ffffff" strokeWidth="1.4"
   618→                  />
   619→                </g>
   620→              )}
   621→
   622→              {/* Time axis labels */}
   623→              {geo.timeLabels.map((tl, i) => (
   624→                <text
   625→                  key={i}
   626→                  x={tl.x}
   627→                  y={H - 6}
   628→                  textAnchor={tl.anchor}
   629→                  fill="var(--gm-text-tertiary)"
   630→                  fillOpacity="0.9"
   631→                  fontSize="8"
   632→                  fontWeight="600"
   633→                  fontFamily="system-ui, sans-serif"
   634→                >
   635→                  {tl.label}
   636→                </text>
   637→              ))}
   638→            </svg>
   639→
   640→            {/* Glass tooltip */}
   641→            {scrubPt && scrubSel && (
   642→              <div
   643→                className="grain-scrub-tooltip"
   644→                style={{ left: scrubSel.leftPx, top: scrubSel.topPx }}
   645→                role="status"
   646→              >
   647→                <span className="v">{scrubPt.v.toFixed(1)}{unitSuffix}</span>
   648→                <span className="t">{format(scrubPt.t, 'MMM d · HH:mm')}</span>
   649→              </div>
   650→            )}
   651→          </div>
   652→
   653→          {/* Per-range mini stats */}
   654→          <div className="grid grid-cols-4 gap-2 mt-3 pt-3 grain-trends-stats grain-tabular">
   655→            <MiniStat
   656→              label={isMoisture ? t('history.stat_avg', lang) : t('history.stat_temp', lang)}
   657→              value={`${geo.avg.toFixed(1)}${unitSuffix}`}
   658→              color={accentHex}
   659→            />
   660→            <MiniStat label={t('history.stat_min', lang)} value={`${geo.min.toFixed(1)}${unitSuffix}`} />
   661→            <MiniStat label={t('history.stat_max', lang)} value={`${geo.max.toFixed(1)}${unitSuffix}`} />
   662→            <MiniStat label={t('history.trends_count', lang)} value={String(geo.pts.length)} />
   663→          </div>
   664→        </>
   665→      )}
   666→    </div>
   667→  );
   668→}
   669→
   670→function MiniStat({ label, value, color }: { label: string; value: string; color?: string }) {
   671→  return (
   672→    <div className="text-center">
   673→      <div className="text-[7.5px] font-bold tracking-[0.12em] uppercase mb-0.5 truncate" style={{ color: 'var(--gm-text-tertiary)' }}>{label}</div>
   674→      <div className="text-[12px] font-bold leading-tight" style={{ color: color ?? 'var(--gm-text-secondary)' }}>{value}</div>
   675→    </div>
   676→  );
   677→}
