    53→    53→interface CrossPt {

    60→    60→interface TrendDir {
    61→    61→  dir: 'up' | 'down' | 'flat';
    62→    62→  delta: number;
    63→    63→  labelKey: string;
    64→    64→}

    66→    66→/** Direction + signed delta across the visible range (per metric) */
    67→    67→function trendDirection(delta: number, metric: MetricId): TrendDir {
    68→    68→  const eps = metric === 'moisture' ? 0.15 : 0.3;
    69→    69→  if (delta > eps) return { dir: 'up', delta, labelKey: 'history.trend_rising' };
    70→    70→  if (delta < -eps) return { dir: 'down', delta, labelKey: 'history.trend_falling' };
    71→    71→  return { dir: 'flat', delta, labelKey: 'history.trend_stable' };
    72→    72→}

    74→    74→function filterEntries(entries: HistoryEntry[], range: RangeId): HistoryEntry[] {
    75→    75→  const r = RANGES.find((x) => x.id === range);
    76→    76→  if (!r || r.ms == null) return entries;
    77→    77→  const cutoff = Date.now() - r.ms;
    78→    78→  return entries.filter((e) => e.reading.timestamp >= cutoff);
    79→    79→}

    81→    81→/** Trailing moving average with a shrinking window at the series head, so the
    82→    82→ *  overlay spans the full chart even for the first win-1 readings. */
    83→    83→function movingAverage(values: number[], win: number): number[] {
    84→    84→  const out: number[] = [];
    85→    85→  let sum = 0;
    86→    86→  for (let i = 0; i < values.length; i++) {
    87→    87→    sum += values[i];
    88→    88→    if (i >= win) sum -= values[i - win];
    89→    89→    out.push(sum / Math.min(i + 1, win));
    90→    90→  }
    91→    91→  return out;
    92→    92→}
    93→    93→
    94→    94→/** Points where the raw curve crosses the MA line — the classic momentum-shift
    95→    95→ *  signal. A crossing counts when the sign flips with at least one side clearly
    96→    96→ *  off the line; when BOTH sides hug the line it is jitter and skipped. */
    97→    97→function detectCrossings(values: number[], ma: number[], xOf: (i: number) => number, yOf: (v: number) => number): CrossPt[] {
    98→    98→  const eps = 0.05;
    99→    99→  const crossings: CrossPt[] = [];
   100→   100→  for (let i = 1; i < values.length; i++) {
   101→   101→    const dPrev = values[i - 1] - ma[i - 1];
   102→   102→    const dCur = values[i] - ma[i];
   103→   103→    if (Math.abs(dPrev) < eps && Math.abs(dCur) < eps) continue;
   104→   104→    if (dPrev < 0 && dCur > 0) crossings.push({ i, x: xOf(i), y: yOf(values[i]), dir: 'up' });
   105→   105→    else if (dPrev > 0 && dCur < 0) crossings.push({ i, x: xOf(i), y: yOf(values[i]), dir: 'down' });
   106→   106→  }
   107→   107→  // Long ranges: subsample so the chart stays readable (keeps most recent)
   108→   108→  const MAX_CROSS = 40;
   109→   109→  if (crossings.length > MAX_CROSS) {
   110→   110→    const step = Math.ceil(crossings.length / MAX_CROSS);
   111→   111→    return crossings.filter((_, k) => k % step === 0);
   112→   112→  }
   113→   113→  return crossings;
   114→   114→}

   116→   116→/** Smallest range with ≥2 points, falling back to ALL */
   117→   117→function detectRange(entries: HistoryEntry[]): RangeId {
   118→   118→  for (const r of RANGES) {
   119→   119→    if (filterEntries(entries, r.id).length >= 2) return r.id;
   120→   120→  }
   121→   121→  return 'all';
   122→   122→}

   124→   124→/** Monotone cubic Bézier (Fritsch–Carlson) — shared curve algorithm */
   125→   125→function monotoneCubic(pts: { x: number; y: number }[]) {
   126→   126→  const n = pts.length;
   127→   127→  if (n < 2) return [];
   128→   128→  const dx: number[] = [];
   129→   129→  const dy: number[] = [];
   130→   130→  const m: number[] = [];
   131→   131→  for (let i = 0; i < n - 1; i++) {
   132→   132→    dx.push(pts[i + 1].x - pts[i].x);
   133→   133→    dy.push(pts[i + 1].y - pts[i].y);
   134→   134→    m.push(dy[i] / (dx[i] || 1));
   135→   135→  }
   136→   136→  const c: number[] = [m[0]];
   137→   137→  for (let i = 1; i < n - 1; i++) {
   138→   138→    if (m[i - 1] * m[i] <= 0) c.push(0);
   139→   139→    else {
   140→   140→      const common = dx[i - 1] + dx[i];
   141→   141→      c.push((3 * common) / ((common + dx[i]) / m[i - 1] + (common + dx[i - 1]) / m[i]));
   142→   142→    }
   143→   143→  }
   144→   144→  c.push(m[n - 2]);
   145→   145→  for (let i = 0; i < n - 1; i++) {
   146→   146→    const acc = c[i] / (m[i] || 1);
   147→   147→    const succ = c[i + 1] / (m[i] || 1);
   148→   148→    if (acc < 0 || succ < 0) {
   149→   149→      c[i] = 0;
   150→   150→      c[i + 1] = 0;
   151→   151→    }
   152→   152→  }
   153→   153→  const segs: { x: number; y: number; cp1x: number; cp1y: number; cp2x: number; cp2y: number }[] = [];
   154→   154→  for (let i = 0; i < n - 1; i++) {
   155→   155→    const p0 = pts[i];
   156→   156→    const p1 = pts[i + 1];
   157→   157→    const segDx = dx[i];
   158→   158→    segs.push({
   159→   159→      x: p1.x, y: p1.y,
   160→   160→      cp1x: p0.x + segDx / 3, cp1y: p0.y + (segDx / 3) * c[i],
   161→   161→      cp2x: p1.x - segDx / 3, cp2y: p1.y - (segDx / 3) * c[i + 1],
   162→   162→    });
   163→   163→  }
   164→   164→  return segs;
   165→   165→}

   167→   167→export function HistoryTrends({ entries }: { entries: HistoryEntry[] }) {
   168→   168→  const { settings, updateSettings } = useGrainStore();
   169→   169→  const lang = settings.language as AppLanguage;
   170→   170→  const accentHex = getAccentHex(settings.accentColor);
   171→   171→  const { safe, warn, critical } = settings.thresholds;
   172→   172→
   173→   173→  // Moving-average window (persisted; falls back to 7 for out-of-range values)
   174→   174→  const maWin = MA_WINDOW_OPTIONS.includes(settings.trendsMaWindow) ? settings.trendsMaWindow : MA_DEFAULT;
   175→   175→
   176→   176→  // Metric + range persist across sessions via AppSettings ('auto' re-detects)
   177→   177→  const [rangeSel, setRangeSel] = useState<RangeId>(() =>
   178→   178→    settings.trendsRange === 'auto' ? detectRange(entries) : settings.trendsRange,
   179→   179→  );
   180→   180→  const [metric, setMetric] = useState<MetricId>(settings.trendsMetric);
   181→   181→
   182→   182→  const filtered = useMemo(() => filterEntries(entries, rangeSel), [entries, rangeSel]);
   183→   183→
   184→   184→  /* ── Geometry ── */
   185→   185→  const geo = useMemo(() => {
   186→   186→    if (filtered.length < 2) return null;
   187→   187→
   188→   188→    // Ascending by time
   189→   189→    const sorted = [...filtered].sort((a, b) => a.reading.timestamp - b.reading.timestamp);
   190→   190→    const getV = (e: HistoryEntry) =>
   191→   191→      metric === 'moisture' ? e.reading.moisture : e.reading.temperature;
   192→   192→    // Temperature is required for temp mode — tolerate legacy entries without it
   193→   193→    const usable = metric === 'temp'
   194→   194→      ? sorted.filter((e) => typeof e.reading.temperature === 'number')
   195→   195→      : sorted;
   196→   196→    if (usable.length < 2) return null;
   197→   197→
   198→   198→    const values = usable.map(getV);
   199→   199→
   200→   200→    const tMin = usable[0].reading.timestamp;
   201→   201→    const tMax = usable[usable.length - 1].reading.timestamp;
   202→   202→    const tRange = tMax - tMin || 1;
   203→   203→
   204→   204→    let lo: number;
   205→   205→    let hi: number;
   206→   206→    let vPad: number;
   207→   207→    if (metric === 'moisture') {
   208→   208→      // Include thresholds in the domain so zone bands are always visible
   209→   209→      lo = Math.min(...values, safe);
   210→   210→      hi = Math.max(...values, critical);
   211→   211→      vPad = Math.max((hi - lo) * 0.25, 1.2);
   212→   212→    } else {
   213→   213→      lo = Math.min(...values);
   214→   214→      hi = Math.max(...values);
   215→   215→      vPad = Math.max((hi - lo) * 0.3, 1.5);
   216→   216→    }
   217→   217→    const vLo = lo - vPad;
   218→   218→    const vHi = hi + vPad;
   219→   219→    const vSpan = vHi - vLo;
   220→   220→
   221→   221→    const drawW = W - PAD_X * 2;
   222→   222→    const drawH = H - PAD_TOP - PAD_BOT;
   223→   223→
   224→   224→    const xOf = (ts: number) => PAD_X + ((ts - tMin) / tRange) * drawW;
   225→   225→    const yOf = (v: number) => PAD_TOP + drawH - ((v - vLo) / vSpan) * drawH;
   226→   226→
   227→   227→    const pts: TrendPt[] = usable.map((e) => ({
   228→   228→      x: xOf(e.reading.timestamp),
   229→   229→      y: yOf(getV(e)),
   230→   230→      v: getV(e),
   231→   231→      t: e.reading.timestamp,
   232→   232→      state: e.decision.state,
   233→   233→    }));
   234→   234→
   235→   235→    // Trailing moving average overlay (same x coordinates as the raw curve)
   236→   236→    const maValues = movingAverage(values, maWin);
   237→   237→    const maPts = pts.map((p, i) => ({ x: p.x, y: yOf(maValues[i]) }));
   238→   238→    let maPathD = '';
   239→   239→    if (maPts.length >= 2) {
   240→   240→      maPathD = `M${maPts[0].x.toFixed(2)},${maPts[0].y.toFixed(2)}`;
   241→   241→      for (const seg of monotoneCubic(maPts)) {
   242→   242→        maPathD += ` C${seg.cp1x.toFixed(2)},${seg.cp1y.toFixed(2)} ${seg.cp2x.toFixed(2)},${seg.cp2y.toFixed(2)} ${seg.x.toFixed(2)},${seg.y.toFixed(2)}`;
   243→   243→      }
   244→   244→    }
   245→   245→
   246→   246→    // Momentum-shift markers: raw curve crossing the MA line
   247→   247→    const crossings = settings.trendsMovingAvg
   248→   248→      ? detectCrossings(values, maValues, (i) => pts[i].x, yOf)
   249→   249→      : [];
   250→   250→
   251→   251→    // Curve path
   252→   252→    let pathD = `M${pts[0].x.toFixed(2)},${pts[0].y.toFixed(2)}`;
   253→   253→    for (const seg of monotoneCubic(pts)) {
   254→   254→      pathD += ` C${seg.cp1x.toFixed(2)},${seg.cp1y.toFixed(2)} ${seg.cp2x.toFixed(2)},${seg.cp2y.toFixed(2)} ${seg.x.toFixed(2)},${seg.y.toFixed(2)}`;
   255→   255→    }
   256→   256→    const fillD = `${pathD} L${pts[pts.length - 1].x.toFixed(2)},${H - PAD_BOT} L${pts[0].x.toFixed(2)},${H - PAD_BOT} Z`;
   257→   257→
   258→   258→    // Zone bands (clamped to the plot area) — moisture only (thresholds are moisture-specific)
   259→   259→    const plotBottom = H - PAD_BOT;
   260→   260→    const band = (vFrom: number, vTo: number) => {
   261→   261→      const yA = Math.max(PAD_TOP, Math.min(plotBottom, yOf(vFrom)));
   262→   262→      const yB = Math.max(PAD_TOP, Math.min(plotBottom, yOf(vTo)));
   263→   263→      return { y: Math.min(yA, yB), h: Math.abs(yB - yA) };
   264→   264→    };
   265→   265→    const critBand = band(Math.max(critical, vLo), vHi);        // above critical
   266→   266→    const warnBand = band(safe, critical);                       // between safe and critical
   267→   267→    const safeBand = band(vLo, safe);                            // below safe
   268→   268→
   269→   269→    // Stats
   270→   270→    const avg = values.reduce((s, v) => s + v, 0) / values.length;
   271→   271→    const min = Math.min(...values);
   272→   272→    const max = Math.max(...values);
   273→   273→
   274→   274→    // Dot thinning for long ranges
   275→   275→    const dotStep = pts.length > 60 ? Math.ceil(pts.length / 48) : 1;
   276→   276→
   277→   277→    // Time axis labels: first / mid / last
   278→   278→    const midIdx = Math.floor(pts.length / 2);
   279→   279→    const fmt = (ts: number) => (rangeSel === '24h' ? format(ts, 'HH:mm') : format(ts, 'MMM d'));
   280→   280→    const timeLabels = [
   281→   281→      { x: pts[0].x, anchor: 'start' as const, label: fmt(tMin) },
   282→   282→      ...(pts.length > 6 ? [{ x: pts[midIdx].x, anchor: 'middle' as const, label: fmt(usable[midIdx].reading.timestamp) }] : []),
   283→   283→      { x: pts[pts.length - 1].x, anchor: 'end' as const, label: fmt(tMax) },
   284→   284→    ];
   285→   285→
   286→   286→    return {
   287→   287→      pts, pathD, fillD, critBand, warnBand, safeBand, avg, min, max,
   288→   288→      avgY: yOf(avg), dotStep, timeLabels,
   289→   289→      safeY: yOf(safe), criticalY: yOf(critical),
   290→   290→      trend: trendDirection(values[values.length - 1] - values[0], metric),
   291→   291→      maPathD, maValues, crossings,
   292→   292→    };
   293→   293→  }, [filtered, safe, critical, rangeSel, metric, maWin, settings.trendsMovingAvg]);
   294→   294→
   295→   295→  /* ── Scrub state (timestamp-anchored, same pattern as hero sparkline) ── */
   296→   296→  const [scrubSel, setScrubSel] = useState<{ t: number; leftPx: number; topPx: number } | null>(null);
   297→   297→  const svgRef = useRef<SVGSVGElement | null>(null);
   298→   298→  const wrapRef = useRef<HTMLDivElement | null>(null);
   299→   299→  const draggingRef = useRef(false);
   300→   300→  const autoClearRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   301→   301→
   302→   302→  const scrubPt = useMemo(
   303→   303→    () => (scrubSel && geo ? geo.pts.find((p) => p.t === scrubSel.t) ?? null : null),
   304→   304→    [geo, scrubSel],
   305→   305→  );
   306→   306→  const scrubIdx = useMemo(
   307→   307→    () => (geo && scrubSel ? geo.pts.findIndex((p) => p.t === scrubSel.t) : -1),
   308→   308→    [geo, scrubSel],
   309→   309→  );
   310→   310→
   311→   311→  const clearAutoTimer = useCallback(() => {
   312→   312→    if (autoClearRef.current) {
   313→   313→      clearTimeout(autoClearRef.current);
   314→   314→      autoClearRef.current = null;
   315→   315→    }
   316→   316→  }, []);
   317→   317→
   318→   318→  const clearScrub = useCallback(() => {
   319→   319→    clearAutoTimer();
   320→   320→    draggingRef.current = false;
   321→   321→    setScrubSel(null);
   322→   322→  }, [clearAutoTimer]);
   323→   323→
   324→   324→  useEffect(() => () => clearAutoTimer(), [clearAutoTimer]);
   325→   325→
   326→   326→  /** Switch range or metric: clears any active scrub (selection is scope-scoped)
   327→   327→   *  and persists the choice so the chart reopens in the same configuration. */
   328→   328→  const selectRange = useCallback(
   329→   329→    (id: RangeId) => {

   332→   332→      setScrubSel(null);

   336→   336→    [clearAutoTimer, updateSettings],

   346→   346→    },
   347→   347→    [clearAutoTimer, updateSettings],
   348→   348→  );
   349→   349→
   350→   350→  /** MA window select / MA toggle — persists like metric + range. */

   354→   354→      updateSettings({ trendsMaWindow: w, trendsMovingAvg: true });
   355→   355→    },
   356→   356→    [settings.hapticsEnabled, updateSettings],

   363→   363→
   364→   364→  /** Index set of crossing points, for the scrub tooltip crossover row */
   365→   365→  const crossByIdx = useMemo(() => {
   366→   366→    const m = new Map<number, CrossPt>();
   367→   367→    if (geo) for (const c of geo.crossings) m.set(c.i, c);
   368→   368→    return m;
   369→   369→  }, [geo]);
   370→   370→
   371→   371→  const anchorFor = useCallback(
   372→   372→    (i: number): { leftPx: number; topPx: number } | null => {
   373→   373→      const svg = svgRef.current;
   374→   374→      const wrap = wrapRef.current;
   375→   375→      if (!svg || !wrap || !geo || i < 0 || i >= geo.pts.length) return null;
   376→   376→      const ctm = svg.getScreenCTM();
   377→   377→      if (!ctm) return null;
   378→   378→      const pt = geo.pts[i];
   379→   379→      let cx: number;
   380→   380→      let cy: number;
   381→   381→      if (typeof DOMPoint === 'function') {
   382→   382→        const sp = new DOMPoint(pt.x, pt.y).matrixTransform(ctm);
   383→   383→        cx = sp.x; cy = sp.y;
   384→   384→      } else {
   385→   385→        cx = pt.x * ctm.a + pt.y * ctm.c + ctm.e;
   386→   386→        cy = pt.x * ctm.b + pt.y * ctm.d + ctm.f;
   387→   387→      }
   388→   388→      const rect = wrap.getBoundingClientRect();
   389→   389→      const leftPx = Math.min(Math.max(cx - rect.left, TOOLTIP_CLAMP), Math.max(rect.width - TOOLTIP_CLAMP, TOOLTIP_CLAMP));
   390→   390→      const topPx = cy - rect.top;
   391→   391→      return { leftPx, topPx };
   392→   392→    },
   393→   393→    [geo],
   394→   394→  );
   395→   395→
   396→   396→  const nearestIndex = useCallback(
   397→   397→    (vx: number): number => {
   398→   398→      if (!geo || geo.pts.length === 0) return -1;
   399→   399→      const pts = geo.pts;
   400→   400→      let lo = 0;
   401→   401→      let hi = pts.length - 1;
   402→   402→      while (hi - lo > 1) {
   403→   403→        const mid = (lo + hi) >> 1;
