    15→function getTrendIcon(trend: string) {
    16→  switch (trend) {
    17→    case 'rising': case 'spike': case 'rising_rapidly': return <TrendUp size={14} weight="bold" />;
    18→    case 'falling': case 'drop': case 'falling_rapidly': return <TrendDown size={14} weight="bold" />;
    19→    default: return <Minus size={14} weight="bold" />;
    20→  }
    21→}

    23→/** Compact human time span: "45m", "2h 30m", "3d 4h" */
    24→function formatSpan(ms: number): string {
    25→  const minutes = Math.round(ms / 60000);
    26→  if (minutes < 1) return '—';
    27→  if (minutes < 90) return `${minutes}m`;
    28→  const hours = Math.floor(minutes / 60);
    29→  if (hours < 48) return `${hours}h ${minutes % 60}m`;
    30→  const days = Math.floor(hours / 24);
    31→  return `${days}d ${hours % 24}h`;
    32→}

    34→/* ═══ Delta vs previous reading ═══
    35→   Storage semantics: rising moisture is the risk direction
    36→   (amber), falling is recovering (green), ±<0.05 is flat. */
    37→
    38→interface Delta {
    39→  dir: 'up' | 'down' | 'flat';
    40→  value: number;
    41→  color: string;
    42→}
    43→
    44→function deltaColor(dir: Delta['dir']): string {
    45→  if (dir === 'up') return 'var(--gm-up)';
    46→  if (dir === 'down') return 'var(--gm-down)';
    47→  return 'var(--gm-text-tertiary)';
    48→}
    49→
    50→function computeDelta(entry: HistoryEntry, prev: HistoryEntry | undefined): Delta | null {
    51→  if (!prev) return null;
    52→  const d = entry.reading.moisture - prev.reading.moisture;
    53→  if (Math.abs(d) < 0.05) return { dir: 'flat', value: d, color: deltaColor('flat') };
    54→  return { dir: d > 0 ? 'up' : 'down', value: d, color: deltaColor(d > 0 ? 'up' : 'down') };
    55→}

    57→function DeltaChip({ delta, ariaLabel }: { delta: Delta; ariaLabel: string }) {
    58→  return (
    59→    <span className="grain-delta-chip" style={{ color: delta.color }} role="status" aria-label={ariaLabel}>
    60→      {delta.dir === 'up'
    61→        ? <TrendUp size={10} weight="bold" />
    62→        : delta.dir === 'down'
    63→          ? <TrendDown size={10} weight="bold" />
    64→          : <Minus size={10} weight="bold" />}
    65→      {(delta.value > 0 ? '+' : '') + delta.value.toFixed(1)}%
    66→    </span>
    67→  );
    68→}

    70→interface Stats {
    71→  avg: number;
    72→  min: number;
    73→  max: number;
    74→  avgTemp: number;
    75→  count: number;
    76→  spanMs: number;
    77→}
    78→
    79→function computeStats(entries: HistoryEntry[]): Stats | null {
    80→  if (entries.length === 0) return null;
    81→  let sum = 0;
    82→  let min = Number.POSITIVE_INFINITY;
    83→  let max = Number.NEGATIVE_INFINITY;
    84→  let tempSum = 0;
    85→  let newest = 0;
    86→  let oldest = Number.POSITIVE_INFINITY;
    87→
    88→  for (const e of entries) {
    89→    const m = e.reading.moisture;
    90→    sum += m;
    91→    if (m < min) min = m;
    92→    if (m > max) max = m;
    93→    tempSum += e.reading.temperature;
    94→    const ts = e.reading.timestamp;
    95→    if (ts > newest) newest = ts;
    96→    if (ts < oldest) oldest = ts;
    97→  }
    98→
    99→  return {
   100→    avg: sum / entries.length,
   101→    min,
   102→    max,
   103→    avgTemp: tempSum / entries.length,
   104→    count: entries.length,
   105→    spanMs: newest - oldest,
   106→  };
   107→}

   113→function HistoryStats({ entries }: { entries: HistoryEntry[] }) {
   114→  const { settings } = useGrainStore();
   115→  const lang = settings.language as AppLanguage;
   116→  const { safe, warn, critical } = settings.thresholds;
   117→
   118→  const stats = useMemo(() => computeStats(entries), [entries]);
   119→  if (!stats) return null;
   120→
   121→  // Zone bar domain: 0 → critical + 4 (stable visual scale)
   122→  const barMax = critical + 4;
   123→  const pct = (v: number) => `${Math.min(100, Math.max(0, (v / barMax) * 100))}%`;
   124→  const safeW = pct(safe);
   125→  const warnW = pct(critical - safe);
   126→  const critW = pct(barMax - critical);
   127→  const avgPos = pct(stats.avg);
   128→  const minPos = pct(stats.min);
   129→  const maxPos = pct(stats.max);
   130→
   131→  return (
   132→    <div className="grain-stats-card mb-4" role="group" aria-label={t('history.stats_title', lang)}>
   133→      {/* Header row: gauge icon + title */}
   134→      <div className="flex items-center gap-2 mb-3">
   135→        <Gauge size={13} weight="bold" style={{ color: 'var(--gm-accent)' }} />
   136→        <span className="text-[9px] font-bold tracking-[0.18em] uppercase" style={{ color: 'var(--gm-text-tertiary)' }}>
   137→          {t('history.stats_title', lang)}
   138→        </span>
   139→        <span className="ml-auto text-[9px] font-semibold tracking-wide" style={{ color: 'var(--gm-text-tertiary)' }}>
   140→          {formatSpan(stats.spanMs)}
   141→        </span>
   142→      </div>
   143→
   144→      {/* Big average + mini stats grid */}
   145→      <div className="flex items-stretch gap-4">
   146→        <div className="shrink-0">
   147→          <div className="text-[9px] font-bold tracking-[0.15em] uppercase mb-1" style={{ color: 'var(--gm-text-secondary)' }}>

   159→          <StatCell label={t('history.stat_max', lang)} value={`${stats.max.toFixed(1)}%`} />

   162→        </div>
   163→      </div>
   164→
   165→      {/* Threshold zone bar with min/avg/max markers */}
   166→      <div className="mt-4">
   167→        <div className="flex justify-between mb-1.5">
   168→          <span className="text-[8px] font-bold tracking-[0.15em] uppercase" style={{ color: 'var(--gm-text-tertiary)' }}>
   169→            {t('history.stat_range', lang)}
   170→          </span>
   171→          <span className="text-[8px] font-semibold grain-tabular" style={{ color: 'var(--gm-text-tertiary)' }}>
   172→            &lt;{safe}% · &lt;{critical}%
   173→          </span>
   174→        </div>
   175→        <div className="grain-stats-bar" aria-hidden="true">
   176→          <div className="grain-stats-zone" style={{ left: 0, width: safeW, background: 'rgba(34, 197, 94, 0.5)' }} />

   449→function ReadingList() {
   450→  const { historyEntries, setSelectedHistoryId, settings } = useGrainStore();
   451→  const lang = settings.language as AppLanguage;
   452→  const accent = settings.accentColor;
   453→
   454→  if (historyEntries.length === 0) {
   455→    return (
   456→      <div className="flex flex-col items-center justify-center py-16 text-center grain-fade-in">
   457→        <div className="w-16 h-16 rounded-3xl flex items-center justify-center mb-5 grain-card" style={{ transform: 'rotate(-4deg)' }}>
   458→          <ArrowClockwise size={28} weight="bold" style={{ color: 'var(--gm-accent)' }} />
   459→        </div>
   460→        <p className="text-sm font-bold mb-1.5" style={{ color: 'var(--gm-text-primary)' }}>{t('history.no_history', lang)}</p>
   461→        <p className="text-xs max-w-[220px] leading-relaxed" style={{ color: 'var(--gm-text-tertiary)' }}>{t('history.no_history_desc', lang)}</p>
   462→      </div>
   463→    );
   464→  }
   465→
   466→  return (
   467→    <div className="flex flex-col gap-2 grain-fade-in">
   468→      <HistoryStats entries={historyEntries} />
   469→      {historyEntries.length >= 3 && <StorageScoreCard entries={historyEntries} />}
   470→      {historyEntries.length >= 2 && <HistoryTrends entries={historyEntries} />}
   471→      {historyEntries.map((entry, i) => {
   472→        const dotHex = getRiskColor(entry.decision.state, accent);
   473→        // Entries are newest-first → the previous chronological reading sits one slot down
   474→        const delta = computeDelta(entry, historyEntries[i + 1]);
   475→        return (

   500→function ReadingDetail({ entry, onClose }: { entry: HistoryEntry; onClose: () => void }) {
   501→  const { reading, decision } = entry;
   502→  const { settings, historyEntries, showToast } = useGrainStore();
   503→  const lang = settings.language as AppLanguage;
   504→  const accent = settings.accentColor;
   505→  const dotHex = getRiskColor(decision.state, accent);
   506→  const [shared, setShared] = useState(false);
   507→
   508→  // Use i18n keys if available, fallback to stored English
   509→  const messageText = decision.messageKey ? t(decision.messageKey, lang) : decision.message;
   510→  const actionText = decision.actionKey ? t(decision.actionKey, lang) : decision.action;
   511→  const trendLabel = tTrend(decision.trend, lang);
   512→
   513→  // Delta vs the previous chronological reading (list is newest-first)
   514→  const idx = historyEntries.findIndex((e) => e.reading.id === reading.id);
   515→  const prev = idx >= 0 ? historyEntries[idx + 1] : undefined;
   516→  const delta = computeDelta(entry, prev);
   517→
   518→  const stateLabel =
   519→    decision.state === 'safe' ? t('history.safe', lang)
   520→    : decision.state === 'warning' ? t('history.warning', lang)
   521→    : decision.state === 'critical' ? t('history.critical', lang)
   522→    : decision.state === 'monitor' ? t('status.monitor', lang)
   523→    : decision.state === 'recovery' ? t('status.recovery', lang)
   524→    : decision.state === 'insufficient_data' ? t('status.learning', lang)
   525→    : decision.state === 'invalid' ? t('status.invalid', lang)
   526→    : decision.state;
   527→
   528→  /* Share via the Web Share API when available (PWA-friendly),
   529→     falling back to clipboard copy. User-cancel is silent. */
   530→  const shareReading = async () => {
   531→    haptic(settings.hapticsEnabled, 8);
   532→    const text =
   533→      `Grain Monitor — ${tGrain(reading.grainType, lang)}\n` +
   534→      `${t('history.moisture', lang)}: ${reading.moisture.toFixed(1)}% · ${t('history.temperature', lang)}: ${Math.round(reading.temperature)}°C\n` +
   535→      `${format(reading.timestamp, 'MMM d, yyyy · h:mm a')}\n` +
   536→      `${stateLabel} · ${messageText}`;
   537→    // Cast mirrors bluetooth.ts: web-bluetooth Navigator typing breaks
   538→    // `in`-narrowing for share(), so we model the optional members directly.
   539→    const navExt = (typeof navigator !== 'undefined' ? navigator : undefined) as
   540→      | (Navigator & { share?: (data?: ShareData) => Promise<void> })
   541→      | undefined;
   542→    try {
   543→      if (navExt?.share) {
   544→        await navExt.share({ title: 'Grain Monitor', text });
   545→        return;
   546→      }
   547→      if (navExt?.clipboard?.writeText) {
   548→        await navExt.clipboard.writeText(text);
   549→        showToast(t('toast.copied', lang));
   550→        setShared(true);
   551→        setTimeout(() => setShared(false), 1600);
   552→        return;
   553→      }
   554→      showToast(t('toast.share_fail', lang));
   555→    } catch (err) {
   556→      // AbortError = the user dismissed the share sheet — not a failure
   557→      if (err instanceof DOMException && err.name === 'AbortError') return;
   558→      showToast(t('toast.share_fail', lang));
   559→    }
   560→  };
   561→
   562→  return (
   563→    <div className="grain-fade-in">
   564→      <div className="flex items-center justify-between mb-5">
   565→        <div>
   566→          <h3 className="text-lg font-bold tracking-tight" style={{ color: 'var(--gm-text-primary)' }}>{t('history.detail', lang)}</h3>
   567→          <p className="text-xs grain-tabular" style={{ color: 'var(--gm-text-secondary)' }}>{format(reading.timestamp, 'EEEE, MMMM d, yyyy · h:mm a')}</p>
   568→        </div>
   569→        <div className="flex items-center gap-2">
   570→          <button onClick={shareReading} className="grain-share-btn" aria-label={t('history.share', lang)}>
   571→            {shared
   572→              ? <Check size={15} weight="bold" />
   573→              : <ShareFat size={15} weight="bold" />}
   574→          </button>
   575→          <button onClick={onClose} className="w-8 h-8 rounded-lg flex items-center justify-center grain-card" style={{ color: 'var(--gm-text-secondary)' }} aria-label={t('history.detail', lang)}><X size={16} weight="bold" /></button>
   576→        </div>
   577→      </div>
   578→      <div className="grid grid-cols-2 gap-3 mb-4">
   579→        <div className="grain-card p-4">
   580→          <div className="text-[9px] font-bold tracking-[0.15em] uppercase mb-2" style={{ color: 'var(--gm-text-secondary)' }}>{t('history.moisture', lang)}</div>
   581→          <div className="flex items-baseline gap-1.5 grain-tabular">
   582→            <span className="text-3xl font-bold" style={{ color: dotHex }}>{reading.moisture.toFixed(1)}</span><span className="text-xs" style={{ color: 'var(--gm-text-secondary)' }}>%</span>
   583→            {delta && <DeltaChip delta={delta} ariaLabel={`${t('history.delta_prev', lang)} ${(delta.value > 0 ? '+' : '') + delta.value.toFixed(1)}%`} />}
   584→          </div>
   585→        </div>
   586→        <div className="grain-card p-4">
   587→          <div className="text-[9px] font-bold tracking-[0.15em] uppercase mb-2" style={{ color: 'var(--gm-text-secondary)' }}>{t('history.temperature', lang)}</div>
   588→          <div className="flex items-baseline gap-1 grain-tabular"><span className="text-3xl font-bold" style={{ color: 'var(--gm-text-primary)' }}>{Math.round(reading.temperature)}</span><span className="text-xs" style={{ color: 'var(--gm-text-secondary)' }}>°C</span></div>
   589→        </div>
   590→      </div>
   591→      <div className="grid grid-cols-3 gap-3 mb-5">
   592→        <div className="grain-card p-3"><div className="text-[9px] font-bold tracking-[0.15em] uppercase mb-1" style={{ color: 'var(--gm-text-secondary)' }}>{t('history.battery', lang)}</div><span className="text-sm font-bold grain-tabular" style={{ color: 'var(--gm-text-primary)' }}>{reading.battery}%</span></div>
   593→        <div className="grain-card p-3"><div className="text-[9px] font-bold tracking-[0.15em] uppercase mb-1" style={{ color: 'var(--gm-text-secondary)' }}>{t('history.signal', lang)}</div><span className="text-sm font-bold grain-tabular" style={{ color: 'var(--gm-text-primary)' }}>{reading.signal}%</span></div>
   594→        <div className="grain-card p-3"><div className="text-[9px] font-bold tracking-[0.15em] uppercase mb-1" style={{ color: 'var(--gm-text-secondary)' }}>{t('history.grain', lang)}</div><span className="text-sm font-bold" style={{ color: 'var(--gm-text-primary)' }}>{tGrain(reading.grainType, lang)}</span></div>
   595→      </div>
   596→      <div className="grain-insight-card mb-5">
   597→        <h4 className="text-xs font-bold tracking-wide uppercase mb-2" style={{ color: dotHex }}>{stateLabel} · {trendLabel}</h4>
   598→        <p className="text-sm leading-relaxed" style={{ color: 'var(--gm-text-secondary)' }}>{messageText}</p>
   599→        <p className="text-[11px] mt-2 leading-relaxed" style={{ color: 'var(--gm-text-tertiary)' }}>{actionText}</p>
   600→      </div>
   601→      <div className="grain-card p-4">
   602→        <div className="flex justify-between items-center mb-2"><span className="text-[10px] font-bold tracking-wider uppercase" style={{ color: 'var(--gm-text-tertiary)' }}>{t('history.rule', lang)}</span><span className="text-[11px] font-mono" style={{ color: 'var(--gm-text-secondary)' }}>{decision.ruleId}</span></div>
   603→        <div className="flex justify-between items-center mb-2"><span className="text-[10px] font-bold tracking-wider uppercase" style={{ color: 'var(--gm-text-tertiary)' }}>{t('history.confidence', lang)}</span><span className="text-[11px] font-bold" style={{ color: 'var(--gm-text-primary)' }}>{decision.confidence.toUpperCase()}</span></div>
   604→        <div className="flex justify-between items-center"><span className="text-[10px] font-bold tracking-wider uppercase" style={{ color: 'var(--gm-text-tertiary)' }}>{t('history.severity', lang)}</span><span className="text-[11px] font-bold grain-tabular" style={{ color: dotHex }}>{decision.severity}/100</span></div>
   605→      </div>
   606→    </div>
   607→  );
   608→}
