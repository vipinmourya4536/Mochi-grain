    16→function getTrendIcon(trend: string) {
    17→  switch (trend) {
    18→    case 'rising': case 'spike': case 'rising_rapidly': return <TrendUp size={14} weight="bold" />;
    19→    case 'falling': case 'drop': case 'falling_rapidly': return <TrendDown size={14} weight="bold" />;
    20→    default: return <Minus size={14} weight="bold" />;
    21→  }
    22→}

    24→/** Compact human time span: "45m", "2h 30m", "3d 4h" */
    25→function formatSpan(ms: number): string {
    26→  const minutes = Math.round(ms / 60000);
    27→  if (minutes < 1) return '—';
    28→  if (minutes < 90) return `${minutes}m`;
    29→  const hours = Math.floor(minutes / 60);
    30→  if (hours < 48) return `${hours}h ${minutes % 60}m`;
    31→  const days = Math.floor(hours / 24);
    32→  return `${days}d ${hours % 24}h`;
    33→}

    35→/* ═══ Delta vs previous reading ═══
    36→   Storage semantics: rising moisture is the risk direction
    37→   (amber), falling is recovering (green), ±<0.05 is flat. */
    38→
    39→interface Delta {
    40→  dir: 'up' | 'down' | 'flat';
    41→  value: number;
    42→  color: string;
    43→}
    44→
    45→function deltaColor(dir: Delta['dir']): string {
    46→  if (dir === 'up') return 'var(--gm-up)';
    47→  if (dir === 'down') return 'var(--gm-down)';
    48→  return 'var(--gm-text-tertiary)';
    49→}
    50→
    51→function computeDelta(entry: HistoryEntry, prev: HistoryEntry | undefined): Delta | null {
    52→  if (!prev) return null;
    53→  const d = entry.reading.moisture - prev.reading.moisture;
    54→  if (Math.abs(d) < 0.05) return { dir: 'flat', value: d, color: deltaColor('flat') };
    55→  return { dir: d > 0 ? 'up' : 'down', value: d, color: deltaColor(d > 0 ? 'up' : 'down') };
    56→}

    58→function DeltaChip({ delta, ariaLabel }: { delta: Delta; ariaLabel: string }) {
    59→  return (
    60→    <span className="grain-delta-chip" style={{ color: delta.color }} role="status" aria-label={ariaLabel}>
    61→      {delta.dir === 'up'
    62→        ? <TrendUp size={10} weight="bold" />
    63→        : delta.dir === 'down'
    64→          ? <TrendDown size={10} weight="bold" />
    65→          : <Minus size={10} weight="bold" />}
    66→      {(delta.value > 0 ? '+' : '') + delta.value.toFixed(1)}%
    67→    </span>
    68→  );

    71→interface Stats {
    72→  avg: number;
    73→  min: number;
    74→  max: number;
    75→  avgTemp: number;
    76→  count: number;
    77→  spanMs: number;
    78→}
    79→
    80→function computeStats(entries: HistoryEntry[]): Stats | null {
    81→  if (entries.length === 0) return null;
    82→  let sum = 0;
    83→  let min = Number.POSITIVE_INFINITY;
    84→  let max = Number.NEGATIVE_INFINITY;
    85→  let tempSum = 0;
    86→  let newest = 0;
    87→  let oldest = Number.POSITIVE_INFINITY;
    88→
    89→  for (const e of entries) {
    90→    const m = e.reading.moisture;
    91→    sum += m;
    92→    if (m < min) min = m;
    93→    if (m > max) max = m;
    94→    tempSum += e.reading.temperature;
    95→    const ts = e.reading.timestamp;
    96→    if (ts > newest) newest = ts;
    97→    if (ts < oldest) oldest = ts;
    98→  }
    99→
   100→  return {
   101→    avg: sum / entries.length,
   102→    min,
   103→    max,
   104→    avgTemp: tempSum / entries.length,
   105→    count: entries.length,
   106→    spanMs: newest - oldest,
   107→  };
   108→}

   124→  const pct = (v: number) => `${Math.min(100, Math.max(0, (v / barMax) * 100))}%`;
   125→  const safeW = pct(safe);
   126→  const warnW = pct(critical - safe);
   127→  const critW = pct(barMax - critical);
   128→  const avgPos = pct(stats.avg);
   129→  const minPos = pct(stats.min);
   130→  const maxPos = pct(stats.max);
   131→
   132→  return (

   141→          {formatSpan(stats.spanMs)}

   450→function ReadingList() {
   451→  const { historyEntries, setSelectedHistoryId, settings } = useGrainStore();

   463→      </div>
   464→    );
   465→  }
   466→
   467→  return (
   468→    <div className="flex flex-col gap-2 grain-fade-in">
   469→      <HistoryStats entries={historyEntries} />
   470→      {historyEntries.length >= 3 && <StorageScoreCard entries={historyEntries} />}
   471→      {historyEntries.length >= 2 && <HistoryTrends entries={historyEntries} />}
   472→      {historyEntries.map((entry, i) => {
   473→        const dotHex = getRiskColor(entry.decision.state, accent);
   474→        // Entries are newest-first → the previous chronological reading sits one slot down
   475→        const delta = computeDelta(entry, historyEntries[i + 1]);

   501→function ReadingDetail({ entry, onClose }: { entry: HistoryEntry; onClose: () => void }) {
   502→  const { reading, decision } = entry;
   503→  const { settings, historyEntries, showToast } = useGrainStore();
   504→  const lang = settings.language as AppLanguage;
   505→  const accent = settings.accentColor;
   506→  const dotHex = getRiskColor(decision.state, accent);
   507→  const [shared, setShared] = useState(false);
   508→  // Context-sparkline metric — moisture is the primary series, temp is one tap away
   509→  const [sparkMetric, setSparkMetric] = useState<'moisture' | 'temp'>('moisture');
   510→
   511→  // Use i18n keys if available, fallback to stored English
   512→  const messageText = decision.messageKey ? t(decision.messageKey, lang) : decision.message;
   513→  const actionText = decision.actionKey ? t(decision.actionKey, lang) : decision.action;
   514→  const trendLabel = tTrend(decision.trend, lang);
   515→
   516→  // Delta vs the previous chronological reading (list is newest-first)
   517→  const idx = historyEntries.findIndex((e) => e.reading.id === reading.id);
   518→  const prev = idx >= 0 ? historyEntries[idx + 1] : undefined;
   519→  const delta = computeDelta(entry, prev);
   520→
   521→  const stateLabel =
   522→    decision.state === 'safe' ? t('history.safe', lang)
   523→    : decision.state === 'warning' ? t('history.warning', lang)
   524→    : decision.state === 'critical' ? t('history.critical', lang)
   525→    : decision.state === 'monitor' ? t('status.monitor', lang)
   526→    : decision.state === 'recovery' ? t('status.recovery', lang)
   527→    : decision.state === 'insufficient_data' ? t('status.learning', lang)
   528→    : decision.state === 'invalid' ? t('status.invalid', lang)
   529→    : decision.state;

   536→      `Grain Monitor — ${tGrain(reading.grainType, lang)}\n` +

   539→      `${stateLabel} · ${messageText}`;
   540→    // Cast mirrors bluetooth.ts: web-bluetooth Navigator typing breaks
   541→    // `in`-narrowing for share(), so we model the optional members directly.
   542→    const navExt = (typeof navigator !== 'undefined' ? navigator : undefined) as
   543→      | (Navigator & { share?: (data?: ShareData) => Promise<void> })
   544→      | undefined;

   546→      if (navExt?.share) {
   547→        await navExt.share({ title: 'Grain Monitor', text });
   548→        return;
   549→      }
   550→      if (navExt?.clipboard?.writeText) {
   551→        await navExt.clipboard.writeText(text);
   552→        showToast(t('toast.copied', lang));
   553→        setShared(true);
   554→        setTimeout(() => setShared(false), 1600);
   555→        return;
   556→      }
   557→      showToast(t('toast.share_fail', lang));
   558→    } catch (err) {
   559→      // AbortError = the user dismissed the share sheet — not a failure
   560→      if (err instanceof DOMException && err.name === 'AbortError') return;
   561→      showToast(t('toast.share_fail', lang));
   562→    }

   567→      <div className="flex items-center justify-between mb-5">
   568→        <div>
   569→          <h3 className="text-lg font-bold tracking-tight" style={{ color: 'var(--gm-text-primary)' }}>{t('history.detail', lang)}</h3>
   570→          <p className="text-xs grain-tabular" style={{ color: 'var(--gm-text-secondary)' }}>{format(reading.timestamp, 'EEEE, MMMM d, yyyy · h:mm a')}</p>
   571→        </div>
   572→        <div className="flex items-center gap-2">
   573→          <button onClick={shareReading} className="grain-share-btn" aria-label={t('history.share', lang)}>
   574→            {shared
   575→              ? <Check size={15} weight="bold" />
   576→              : <ShareFat size={15} weight="bold" />}

   578→          <button onClick={onClose} className="w-8 h-8 rounded-lg flex items-center justify-center grain-card" style={{ color: 'var(--gm-text-secondary)' }} aria-label={t('history.detail', lang)}><X size={16} weight="bold" /></button>
   579→        </div>
   580→      </div>
   581→      <div className="grid grid-cols-2 gap-3 mb-4">
   582→        <div className="grain-card p-4">
   583→          <div className="text-[9px] font-bold tracking-[0.15em] uppercase mb-2" style={{ color: 'var(--gm-text-secondary)' }}>{t('history.moisture', lang)}</div>
   584→          <div className="flex items-baseline gap-1.5 grain-tabular">
   585→            <span className="text-3xl font-bold" style={{ color: dotHex }}>{reading.moisture.toFixed(1)}</span><span className="text-xs" style={{ color: 'var(--gm-text-secondary)' }}>%</span>

   664→/** Trigger a classic blob download (desktop / share-fallback path) */
   665→function downloadCsv(csv: string, filename: string) {
   666→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8' });
   667→  const url = URL.createObjectURL(blob);
   668→  const a = document.createElement('a');
   669→  a.href = url;
   670→  a.download = filename;
   671→  document.body.appendChild(a);
   672→  a.click();
   673→  a.remove();
   674→  setTimeout(() => URL.revokeObjectURL(url), 1200);
   675→}

   677→export function HistoryTab() {
   678→  const { selectedHistoryEntry, setSelectedHistoryId, historyEntries, clearHistory, settings, showToast } = useGrainStore();
   679→  const lang = settings.language as AppLanguage;
   680→
   681→  /* Build a CSV of every stored reading and hand it to the user.
   682→   * Web Share with a file when the platform supports it (mobile PWA
   683→   * share sheet), otherwise a classic blob download. */
   684→  const exportCsv = () => {
   685→    if (historyEntries.length === 0) return;
   686→    haptic(settings.hapticsEnabled, 8);
   687→    const header = 'timestamp,datetime,grain,moisture_pct,temperature_c,battery_pct,signal_pct,state,trend';
   688→    const rows = [...historyEntries]
   689→      .sort((a, b) => a.reading.timestamp - b.reading.timestamp)
   690→      .map((e) => [
   691→        new Date(e.reading.timestamp).toISOString(),
   692→        format(e.reading.timestamp, 'yyyy-MM-dd HH:mm:ss'),
   693→        e.reading.grainType,
   694→        e.reading.moisture.toFixed(2),
   695→        e.reading.temperature.toFixed(1),
   696→        Math.round(e.reading.battery),
   697→        Math.round(e.reading.signal),
   698→        e.decision.state,
   699→        e.decision.trend,
   700→      ].join(','));
   701→    const csv = `${header}\n${rows.join('\n')}\n`;
   702→    const filename = `mochi-grain-readings-${format(Date.now(), 'yyyy-MM-dd')}.csv`;
   703→    const toastOk = () => showToast(tp('toast.exported', lang, { n: historyEntries.length }));
   704→
   705→    // Cast mirrors shareReading(): Web Share file members are absent from the TS DOM lib
   706→    const navExt = (typeof navigator !== 'undefined' ? navigator : undefined) as
   707→      | (Navigator & {
   708→          canShare?: (data: { files?: File[] }) => boolean;
   709→          share?: (data?: { files?: File[]; title?: string; text?: string }) => Promise<void>;
   710→        })
   711→      | undefined;
   712→    let sharePayload: { files: File[]; title: string; text: string } | null = null;
   713→    try {
   714→      if (navExt?.share && navExt?.canShare) {
   715→        const file = new File([csv], filename, { type: 'text/csv' });
   716→        if (navExt.canShare({ files: [file] })) {
   717→          sharePayload = { files: [file], title: 'Grain Monitor', text: filename };
   718→        }
   719→      }
   720→    } catch { /* canShare can throw on odd inputs — fall through to download */ }
   721→
   722→    if (sharePayload) {
   723→      navExt!.share(sharePayload)
   724→        .then(toastOk)
   725→        .catch((err) => {
   726→          // AbortError = user dismissed the share sheet — not a failure
   727→          if (err instanceof DOMException && err.name === 'AbortError') return;
   728→          downloadCsv(csv, filename);
   729→          toastOk();
   730→        });
   731→      return;
   732→    }
   733→    downloadCsv(csv, filename);
   734→    toastOk();
   735→  };
   736→
   737→  return (
   738→    <div className="pt-2 pb-6 grain-fade-in">
   739→      {!selectedHistoryEntry ? (
   740→        <>
   741→          <div className="flex items-start justify-between mb-5">
   742→            <div>
   743→              <h2 className="text-2xl font-bold tracking-tight mb-1" style={{ color: 'var(--gm-text-primary)' }}>{t('history.title', lang)}</h2>
   744→              <p className="text-sm" style={{ color: 'var(--gm-text-secondary)' }}>{historyEntries.length > 0 ? `${historyEntries.length} ${historyEntries.length !== 1 ? t('history.readings', lang) : t('history.reading', lang)}` : t('history.past', lang)}</p>
   745→            </div>
   746→            {historyEntries.length > 0 && (
   747→              <div className="flex items-center gap-2">
   748→                <button onClick={exportCsv} className="grain-export-btn" aria-label={t('history.export', lang)} title={t('history.export', lang)}>
   749→                  <DownloadSimple size={16} weight="bold" />
   750→                </button>
   751→                <button onClick={clearHistory} className="w-8 h-8 rounded-lg flex items-center justify-center grain-card" style={{ color: 'var(--gm-text-tertiary)' }} aria-label={t('settings.clear_history', lang)}><Trash size={16} weight="bold" /></button>
   752→              </div>
   753→            )}
   754→          </div>
   755→          <ReadingList />
   756→        </>
   757→      ) : (
   758→        <ReadingDetail entry={selectedHistoryEntry} onClose={() => setSelectedHistoryId(null)} />
   759→      )}
   760→    </div>
   761→  );
   762→}
