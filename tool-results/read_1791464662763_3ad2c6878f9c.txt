  1254→/** Get translated string with parameter substitution */
  1255→export function tp(key: string, lang: AppLanguage, params?: Record<string, string | number>): string {
  1256→  let str = t(key, lang);
  1257→  if (params) {
  1258→    for (const [k, v] of Object.entries(params)) {
  1259→      str = str.replace(`{${k}}`, String(v));
  1260→    }
  1261→  }
  1262→  return str;
  1263→}
