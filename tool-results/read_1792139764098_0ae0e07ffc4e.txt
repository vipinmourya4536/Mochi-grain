    17→function getTrendIcon(trend: string) {
    18→  switch (trend) {
    19→    case 'rising': case 'spike': case 'rising_rapidly': return <TrendUp size={14} weight="bold" />;
    20→    case 'falling': case 'drop': case 'falling_rapidly': return <TrendDown size={14} weight="bold" />;
    21→    default: return <Minus size={14} weight="bold" />;
    22→  }
    23→}

    25→/** Compact human time span: "45m", "2h 30m", "3d 4h" */
    26→function formatSpan(ms: number): string {
    27→  const minutes = Math.round(ms / 60000);
    28→  if (minutes < 1) return '—';
    29→  if (minutes < 90) return `${minutes}m`;
    30→  const hours = Math.floor(minutes / 60);
    31→  if (hours < 48) return `${hours}h ${minutes % 60}m`;
    32→  const days = Math.floor(hours / 24);
    33→  return `${days}d ${hours % 24}h`;
    34→}

    36→/* ═══ Delta vs previous reading ═══
    37→   Storage semantics: rising moisture is the risk direction
    38→   (amber), falling is recovering (green), ±<0.05 is flat. */
    39→
    40→interface Delta {
    41→  dir: 'up' | 'down' | 'flat';
    42→  value: number;
    43→  color: string;
    44→}
    45→
    46→function deltaColor(dir: Delta['dir']): string {
    47→  if (dir === 'up') return 'var(--gm-up)';
    48→  if (dir === 'down') return 'var(--gm-down)';
    49→  return 'var(--gm-text-tertiary)';
    50→}
    51→
    52→function computeDelta(entry: HistoryEntry, prev: HistoryEntry | undefined): Delta | null {
    53→  if (!prev) return null;
    54→  const d = entry.reading.moisture - prev.reading.moisture;
    55→  if (Math.abs(d) < 0.05) return { dir: 'flat', value: d, color: deltaColor('flat') };
    56→  return { dir: d > 0 ? 'up' : 'down', value: d, color: deltaColor(d > 0 ? 'up' : 'down') };
    57→}

    59→function DeltaChip({ delta, ariaLabel }: { delta: Delta; ariaLabel: string }) {
    60→  return (
    61→    <span className="grain-delta-chip" style={{ color: delta.color }} role="status" aria-label={ariaLabel}>
    62→      {delta.dir === 'up'

    64→        : delta.dir === 'down'
    65→          ? <TrendDown size={10} weight="bold" />
    66→          : <Minus size={10} weight="bold" />}
    67→      {(delta.value > 0 ? '+' : '') + delta.value.toFixed(1)}%
    68→    </span>

    72→interface Stats {
    73→  avg: number;
    74→  min: number;
    75→  max: number;
    76→  avgTemp: number;
    77→  count: number;
    78→  spanMs: number;
    79→}
    80→
    81→function computeStats(entries: HistoryEntry[]): Stats | null {
    82→  if (entries.length === 0) return null;
    83→  let sum = 0;
    84→  let min = Number.POSITIVE_INFINITY;
    85→  let max = Number.NEGATIVE_INFINITY;
    86→  let tempSum = 0;
    87→  let newest = 0;
    88→  let oldest = Number.POSITIVE_INFINITY;
    89→
    90→  for (const e of entries) {
    91→    const m = e.reading.moisture;
    92→    sum += m;
    93→    if (m < min) min = m;
    94→    if (m > max) max = m;
    95→    tempSum += e.reading.temperature;
    96→    const ts = e.reading.timestamp;
    97→    if (ts > newest) newest = ts;
    98→    if (ts < oldest) oldest = ts;
    99→  }
   100→
   101→  return {
   102→    avg: sum / entries.length,
   103→    min,
   104→    max,
   105→    avgTemp: tempSum / entries.length,
   106→    count: entries.length,
   107→    spanMs: newest - oldest,
   108→  };
   109→}

   127→  const warnW = pct(critical - safe);
   128→  const critW = pct(barMax - critical);
   129→  const avgPos = pct(stats.avg);

   131→  const maxPos = pct(stats.max);

   451→function ReadingList() {
   452→  const { historyEntries, setSelectedHistoryId, settings } = useGrainStore();

   464→      </div>
   465→    );
   466→  }
   467→
   468→  return (
   469→    <div className="flex flex-col gap-2 grain-fade-in">
   470→      <HistoryStats entries={historyEntries} />
   471→      {historyEntries.length >= 3 && <StorageScoreCard entries={historyEntries} />}
   472→      {historyEntries.length >= 2 && <HistoryTrends entries={historyEntries} />}
   473→      {historyEntries.map((entry, i) => {
   474→        const dotHex = getRiskColor(entry.decision.state, accent);
   475→        // Entries are newest-first → the previous chronological reading sits one slot down
   476→        const delta = computeDelta(entry, historyEntries[i + 1]);

   502→function ReadingDetail({ entry, onClose }: { entry: HistoryEntry; onClose: () => void }) {
   503→  const { reading, decision } = entry;
   504→  const { settings, historyEntries, showToast, updateSettings } = useGrainStore();
   505→  const lang = settings.language as AppLanguage;
   506→  const accent = settings.accentColor;
   507→  const dotHex = getRiskColor(decision.state, accent);
   508→  const [shared, setShared] = useState(false);
   509→  // Context-sparkline metric — persisted (settings.detailMetric) like the
   510→  // trends chart metric, so the detail view reopens on the last series used
   511→  const sparkMetric = settings.detailMetric;
   512→  const setSparkMetric = (m: typeof sparkMetric) => updateSettings({ detailMetric: m });
   513→
   514→  // Use i18n keys if available, fallback to stored English
   515→  const messageText = decision.messageKey ? t(decision.messageKey, lang) : decision.message;
   516→  const actionText = decision.actionKey ? t(decision.actionKey, lang) : decision.action;
   517→  const trendLabel = tTrend(decision.trend, lang);
   518→
   519→  // Delta vs the previous chronological reading (list is newest-first)
   520→  const idx = historyEntries.findIndex((e) => e.reading.id === reading.id);
   521→  const prev = idx >= 0 ? historyEntries[idx + 1] : undefined;
   522→  const delta = computeDelta(entry, prev);
   523→
   524→  const stateLabel =
   525→    decision.state === 'safe' ? t('history.safe', lang)
   526→    : decision.state === 'warning' ? t('history.warning', lang)
   527→    : decision.state === 'critical' ? t('history.critical', lang)
   528→    : decision.state === 'monitor' ? t('status.monitor', lang)
   529→    : decision.state === 'recovery' ? t('status.recovery', lang)
   530→    : decision.state === 'insufficient_data' ? t('status.learning', lang)
   531→    : decision.state === 'invalid' ? t('status.invalid', lang)
   532→    : decision.state;
   533→
   534→  /* Share via the Web Share API when available (PWA-friendly),
   535→     falling back to clipboard copy. User-cancel is silent. */
   536→  const shareReading = async () => {
   537→    haptic(settings.hapticsEnabled, 8);
   538→    const text =
   539→      `Grain Monitor — ${tGrain(reading.grainType, lang)}\n` +
   540→      `${t('history.moisture', lang)}: ${reading.moisture.toFixed(1)}% · ${t('history.temperature', lang)}: ${Math.round(reading.temperature)}°C\n` +
   541→      `${format(reading.timestamp, 'MMM d, yyyy · h:mm a')}\n` +
   542→      `${stateLabel} · ${messageText}`;
   543→    // Cast mirrors bluetooth.ts: web-bluetooth Navigator typing breaks
   544→    // `in`-narrowing for share(), so we model the optional members directly.
   545→    const navExt = (typeof navigator !== 'undefined' ? navigator : undefined) as
   546→      | (Navigator & { share?: (data?: ShareData) => Promise<void> })
   547→      | undefined;
   548→    try {
   549→      if (navExt?.share) {
   550→        await navExt.share({ title: 'Grain Monitor', text });
   551→        return;
   552→      }
   553→      if (navExt?.clipboard?.writeText) {
   554→        await navExt.clipboard.writeText(text);
   555→        showToast(t('toast.copied', lang));
   556→        setShared(true);
   557→        setTimeout(() => setShared(false), 1600);
   558→        return;
   559→      }
   560→      showToast(t('toast.share_fail', lang));
   561→    } catch (err) {
   562→      // AbortError = the user dismissed the share sheet — not a failure
   563→      if (err instanceof DOMException && err.name === 'AbortError') return;
   564→      showToast(t('toast.share_fail', lang));
   565→    }
   566→  };
   567→
   568→  return (
   569→    <div className="grain-fade-in">
   570→      <div className="flex items-center justify-between mb-5">
   571→        <div>
   572→          <h3 className="text-lg font-bold tracking-tight" style={{ color: 'var(--gm-text-primary)' }}>{t('history.detail', lang)}</h3>
   573→          <p className="text-xs grain-tabular" style={{ color: 'var(--gm-text-secondary)' }}>{format(reading.timestamp, 'EEEE, MMMM d, yyyy · h:mm a')}</p>
   574→        </div>
   575→        <div className="flex items-center gap-2">
   576→          <button onClick={shareReading} className="grain-share-btn" aria-label={t('history.share', lang)}>
   577→            {shared
   578→              ? <Check size={15} weight="bold" />
   579→              : <ShareFat size={15} weight="bold" />}
   580→          </button>
   581→          <button onClick={onClose} className="w-8 h-8 rounded-lg flex items-center justify-center grain-card" style={{ color: 'var(--gm-text-secondary)' }} aria-label={t('history.detail', lang)}><X size={16} weight="bold" /></button>
   582→        </div>
   583→      </div>
   584→      <div className="grid grid-cols-2 gap-3 mb-4">
   585→        <div className="grain-card p-4">
   586→          <div className="text-[9px] font-bold tracking-[0.15em] uppercase mb-2" style={{ color: 'var(--gm-text-secondary)' }}>{t('history.moisture', lang)}</div>
   587→          <div className="flex items-baseline gap-1.5 grain-tabular">
   588→            <span className="text-3xl font-bold" style={{ color: dotHex }}>{reading.moisture.toFixed(1)}</span><span className="text-xs" style={{ color: 'var(--gm-text-secondary)' }}>%</span>
   589→            {delta && <DeltaChip delta={delta} ariaLabel={`${t('history.delta_prev', lang)} ${(delta.value > 0 ? '+' : '') + delta.value.toFixed(1)}%`} />}
   590→          </div>
   591→        </div>
   592→        <div className="grain-card p-4">
   593→          <div className="text-[9px] font-bold tracking-[0.15em] uppercase mb-2" style={{ color: 'var(--gm-text-secondary)' }}>{t('history.temperature', lang)}</div>
   594→          <div className="flex items-baseline gap-1 grain-tabular"><span className="text-3xl font-bold" style={{ color: 'var(--gm-text-primary)' }}>{Math.round(reading.temperature)}</span><span className="text-xs" style={{ color: 'var(--gm-text-secondary)' }}>°C</span></div>
   595→        </div>
   596→      </div>
   597→      <div className="grid grid-cols-3 gap-3 mb-4">
   598→        <div className="grain-card p-3"><div className="text-[9px] font-bold tracking-[0.15em] uppercase mb-1" style={{ color: 'var(--gm-text-secondary)' }}>{t('history.battery', lang)}</div><span className="text-sm font-bold grain-tabular" style={{ color: 'var(--gm-text-primary)' }}>{reading.battery}%</span></div>
   599→        <div className="grain-card p-3"><div className="text-[9px] font-bold tracking-[0.15em] uppercase mb-1" style={{ color: 'var(--gm-text-secondary)' }}>{t('history.signal', lang)}</div><span className="text-sm font-bold grain-tabular" style={{ color: 'var(--gm-text-primary)' }}>{reading.signal}%</span></div>

   601→      </div>
   602→
   603→      {/* Local trajectory: the ~16 readings surrounding this one */}
   604→      <div className="grain-card p-4 mb-4">
   605→        <div className="flex items-center justify-between mb-2">
   606→          <span className="text-[9px] font-bold tracking-[0.15em] uppercase" style={{ color: 'var(--gm-text-secondary)' }}>

   667→export function HistoryTab() {
   668→  const { selectedHistoryEntry, setSelectedHistoryId, historyEntries, clearHistory, settings, showToast } = useGrainStore();
   669→  const lang = settings.language as AppLanguage;
   670→
   671→  /* Build a CSV of every stored reading (shared client-side builder —
   672→   * the readings live in IndexedDB, so a server route could never read
   673→   * them) and hand it to the user: Web Share with a file when the
   674→   * platform supports it, otherwise a classic blob download.
   675→   * Decision message/action columns follow settings.exportDecisions. */
   676→  const exportCsv = async () => {
   677→    if (historyEntries.length === 0) {
   678→      showToast(t('toast.nothing_to_export', lang));
   679→      return;
   680→    }
   681→    haptic(settings.hapticsEnabled, 8);
   682→    const csv = buildReadingsCsv(historyEntries, lang, settings.exportDecisions);
   683→    const filename = `mochi-grain-readings-${format(Date.now(), 'yyyy-MM-dd')}.csv`;
   684→    const outcome = await deliverCsv(csv, filename);
   685→    if (outcome !== 'cancelled') {
   686→      showToast(tp('toast.exported', lang, { n: historyEntries.length }));
   687→    }
   688→  };
   689→
   690→  return (
   691→    <div className="pt-2 pb-6 grain-fade-in">
   692→      {!selectedHistoryEntry ? (
   693→        <>
   694→          <div className="flex items-start justify-between mb-5">
   695→            <div>
   696→              <h2 className="text-2xl font-bold tracking-tight mb-1" style={{ color: 'var(--gm-text-primary)' }}>{t('history.title', lang)}</h2>
   697→              <p className="text-sm" style={{ color: 'var(--gm-text-secondary)' }}>{historyEntries.length > 0 ? `${historyEntries.length} ${historyEntries.length !== 1 ? t('history.readings', lang) : t('history.reading', lang)}` : t('history.past', lang)}</p>
   698→            </div>
   699→            {historyEntries.length > 0 && (
   700→              <div className="flex items-center gap-2">
   701→                <button onClick={exportCsv} className="grain-export-btn" aria-label={t('history.export', lang)} title={t('history.export', lang)}>
   702→                  <DownloadSimple size={16} weight="bold" />
   703→                </button>
   704→                <button onClick={clearHistory} className="w-8 h-8 rounded-lg flex items-center justify-center grain-card" style={{ color: 'var(--gm-text-tertiary)' }} aria-label={t('settings.clear_history', lang)}><Trash size={16} weight="bold" /></button>
   705→              </div>
   706→            )}
   707→          </div>
   708→          <ReadingList />
   709→        </>
   710→      ) : (
   711→        <ReadingDetail entry={selectedHistoryEntry} onClose={() => setSelectedHistoryId(null)} />
   712→      )}
   713→    </div>
   714→  );
   715→}
