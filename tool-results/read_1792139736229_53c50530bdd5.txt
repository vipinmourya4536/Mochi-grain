   341→function AdvancedSettingsContent() {
   342→  const {
   343→    settings, updateSettings,
   344→    disconnectProbe, sendProbeCommand,
   345→    syncProbeHistory, clearHistory,
   346→    hasDevice, showToast,
   347→    demoMode, enableDemoMode, disableDemoMode,
   348→    demoIntervalSec, setDemoIntervalSec,
   349→    historyEntries,
   350→    loadHistory, loadSettings,
   351→  } = useGrainStore();
   352→
   353→  const lang = settings.language as AppLanguage;
   354→  const hasConnection = hasDevice && !demoMode;
   355→  const fileInputRef = useRef<HTMLInputElement | null>(null);
   356→  const importModeRef = useRef<'merge' | 'replace'>('merge');
   357→  const [confirmReplace, setConfirmReplace] = useState(false);
   358→  const overlayRef = useRef<HTMLDivElement | null>(null);
   359→  const lastFocusRef = useRef<HTMLElement | null>(null);
   360→
   361→  /* Focus trap + focus restore for the destructive confirmation modal.
   362→     Tab/Shift+Tab cycle inside the dialog; Escape closes (capture-phase,
   363→     works even when focus sits on body); on close, focus returns to the
   364→     element that opened the dialog (captured in the click handler — by the
   365→     time the open-effect runs, React has already applied autoFocus to the
   366→     Cancel button, so the effect would otherwise capture the dialog itself). */
   367→  const openReplaceModal = () => {
   368→    lastFocusRef.current = document.activeElement as HTMLElement | null;
   369→    setConfirmReplace(true);
   370→  };
   371→
   372→  useEffect(() => {
   373→    if (!confirmReplace) return;
   374→    const onKey = (e: KeyboardEvent) => {
   375→      if (e.key === 'Escape') {
   376→        e.stopPropagation();
   377→        setConfirmReplace(false);
   378→        return;
   379→      }
   380→      if (e.key !== 'Tab') return;
   381→      const overlay = overlayRef.current;
   382→      if (!overlay) return;
   383→      const focusables = Array.from(
   384→        overlay.querySelectorAll<HTMLElement>(
   385→          'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])',
   386→        ),
   387→      ).filter((el) => !el.hasAttribute('disabled'));
   388→      if (focusables.length === 0) return;
   389→      const first = focusables[0];
   390→      const last = focusables[focusables.length - 1];
   391→      const active = document.activeElement;
   392→      if (e.shiftKey && (active === first || !overlay.contains(active))) {
   393→        e.preventDefault();
   394→        last.focus();
   395→      } else if (!e.shiftKey && (active === last || !overlay.contains(active))) {
   396→        e.preventDefault();
   397→        first.focus();
   398→      }
   399→    };
   400→    document.addEventListener('keydown', onKey, true);
   401→    return () => {
   402→      document.removeEventListener('keydown', onKey, true);
   403→      lastFocusRef.current?.focus?.();
   404→    };
   405→  }, [confirmReplace]);
   406→
   407→  const openImportPicker = (mode: 'merge' | 'replace') => {
   408→    importModeRef.current = mode;
   409→    fileInputRef.current?.click();
   410→  };
   411→
   412→  const handleImportFile = async (e: React.ChangeEvent<HTMLInputElement>) => {
   413→    const file = e.target.files?.[0];
   414→    e.target.value = ''; // allow re-selecting the same file
   415→    if (!file) return;
   416→    const mode = importModeRef.current;
   417→    try {
   418→      const text = await file.text();
   419→      const result = await importBackupJSON(text, mode);
   420→      // Refresh the live store from the restored database
   421→      await loadHistory();
   422→      await loadSettings();
   423→      showToast(`${t(mode === 'replace' ? 'toast.backup_replaced' : 'toast.backup_imported', lang)} · ${result.history}`);
   424→    } catch (err) {
   425→      showToast(err instanceof Error && err.message === 'invalid'
   426→        ? t('toast.import_invalid', lang)
   427→        : t('toast.import_failed', lang));
   428→    }
