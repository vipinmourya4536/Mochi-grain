   530→    function drawSparkline(data) {
   531→        const width = 300, height = 60, padding = 4;
   532→        const max = Math.max(...data), min = Math.min(...data);
   533→        const range = max - min || 1;
   534→
   535→        let path = '';
   536→        let fillPath = '';
   537→
   538→        data.forEach((val, i) => {
   539→            const x = (i / (data.length - 1)) * width;
   540→            const y = height - padding - ((val - min) / range) * (height - padding * 2);
   541→            path += (i === 0 ? 'M' : 'L') + x + ',' + y;
   542→            if (i === 0) fillPath = 'M' + x + ',' + (height) + ' L' + x + ',' + y;
   543→            else fillPath += ' L' + x + ',' + y;
   544→        });
   545→        fillPath += ' L' + width + ',' + height + ' Z';
   546→
   547→        document.getElementById('sparkPath').setAttribute('d', path);
   548→        document.getElementById('sparkFill').setAttribute('d', fillPath);
   549→    }
