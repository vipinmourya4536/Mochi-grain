    26→    26→const RANGES: { id: RangeId; label: string; ms: number | null }[] = [

    28→    28→  { id: '7d', label: '7D', ms: 7 * 24 * 3600e3 },

    33→    33→const SCRUB_AUTO_CLEAR_MS = 3200;

    35→    35→const MA_WINDOW = 7; // trailing moving-average window (readings)
    36→    36→
    37→    37→const W = 358;

    39→    39→const PAD_X = 18;

    43→    43→interface TrendPt {
    44→    44→  x: number;
    45→    45→  y: number;

    51→    51→interface TrendDir {
    52→    52→  dir: 'up' | 'down' | 'flat';
    53→    53→  delta: number;
    54→    54→  labelKey: string;
    55→    55→}
    56→    56→
    57→    57→/** Direction + signed delta across the visible range (per metric) */
    58→    58→function trendDirection(delta: number, metric: MetricId): TrendDir {
    59→    59→  const eps = metric === 'moisture' ? 0.15 : 0.3;
    60→    60→  if (delta > eps) return { dir: 'up', delta, labelKey: 'history.trend_rising' };
    61→    61→  if (delta < -eps) return { dir: 'down', delta, labelKey: 'history.trend_falling' };
    62→    62→  return { dir: 'flat', delta, labelKey: 'history.trend_stable' };
    63→    63→}
    64→    64→
    65→    65→function filterEntries(entries: HistoryEntry[], range: RangeId): HistoryEntry[] {
    66→    66→  const r = RANGES.find((x) => x.id === range);
    67→    67→  if (!r || r.ms == null) return entries;
    68→    68→  const cutoff = Date.now() - r.ms;
    69→    69→  return entries.filter((e) => e.reading.timestamp >= cutoff);
    70→    70→}
    71→    71→
    72→    72→/** Trailing moving average with a shrinking window at the series head, so the
    73→    73→ *  overlay spans the full chart even for the first MA_WINDOW-1 readings. */
    74→    74→function movingAverage(values: number[]): number[] {
    75→    75→  const out: number[] = [];
    76→    76→  let sum = 0;
    77→    77→  for (let i = 0; i < values.length; i++) {
    78→    78→    sum += values[i];
    79→    79→    if (i >= MA_WINDOW) sum -= values[i - MA_WINDOW];
    80→    80→    out.push(sum / Math.min(i + 1, MA_WINDOW));
    81→    81→  }
    82→    82→  return out;
    83→    83→}
    84→    84→
    85→    85→/** Smallest range with ≥2 points, falling back to ALL */
    86→    86→function detectRange(entries: HistoryEntry[]): RangeId {
    87→    87→  for (const r of RANGES) {
    88→    88→    if (filterEntries(entries, r.id).length >= 2) return r.id;
    89→    89→  }
    90→    90→  return 'all';
    91→    91→}

    93→    93→/** Monotone cubic Bézier (Fritsch–Carlson) — shared curve algorithm */
    94→    94→function monotoneCubic(pts: { x: number; y: number }[]) {
    95→    95→  const n = pts.length;
    96→    96→  if (n < 2) return [];
    97→    97→  const dx: number[] = [];
    98→    98→  const dy: number[] = [];
    99→    99→  const m: number[] = [];
   100→   100→  for (let i = 0; i < n - 1; i++) {
   101→   101→    dx.push(pts[i + 1].x - pts[i].x);
   102→   102→    dy.push(pts[i + 1].y - pts[i].y);
   103→   103→    m.push(dy[i] / (dx[i] || 1));
   104→   104→  }
   105→   105→  const c: number[] = [m[0]];
   106→   106→  for (let i = 1; i < n - 1; i++) {
   107→   107→    if (m[i - 1] * m[i] <= 0) c.push(0);
   108→   108→    else {
   109→   109→      const common = dx[i - 1] + dx[i];
   110→   110→      c.push((3 * common) / ((common + dx[i]) / m[i - 1] + (common + dx[i - 1]) / m[i]));
   111→   111→    }
   112→   112→  }
   113→   113→  c.push(m[n - 2]);
   114→   114→  for (let i = 0; i < n - 1; i++) {
   115→   115→    const acc = c[i] / (m[i] || 1);
   116→   116→    const succ = c[i + 1] / (m[i] || 1);
   117→   117→    if (acc < 0 || succ < 0) {
   118→   118→      c[i] = 0;
   119→   119→      c[i + 1] = 0;
   120→   120→    }
   121→   121→  }
   122→   122→  const segs: { x: number; y: number; cp1x: number; cp1y: number; cp2x: number; cp2y: number }[] = [];
   123→   123→  for (let i = 0; i < n - 1; i++) {
   124→   124→    const p0 = pts[i];
   125→   125→    const p1 = pts[i + 1];
   126→   126→    const segDx = dx[i];
   127→   127→    segs.push({
   128→   128→      x: p1.x, y: p1.y,
   129→   129→      cp1x: p0.x + segDx / 3, cp1y: p0.y + (segDx / 3) * c[i],
   130→   130→      cp2x: p1.x - segDx / 3, cp2y: p1.y - (segDx / 3) * c[i + 1],
   131→   131→    });
   132→   132→  }
   133→   133→  return segs;
   134→   134→}

   136→   136→export function HistoryTrends({ entries }: { entries: HistoryEntry[] }) {
   137→   137→  const { settings, updateSettings } = useGrainStore();
   138→   138→  const lang = settings.language as AppLanguage;
   139→   139→  const accentHex = getAccentHex(settings.accentColor);
   140→   140→  const { safe, warn, critical } = settings.thresholds;
   141→   141→
   142→   142→  // Metric + range persist across sessions via AppSettings ('auto' re-detects)
   143→   143→  const [rangeSel, setRangeSel] = useState<RangeId>(() =>
   144→   144→    settings.trendsRange === 'auto' ? detectRange(entries) : settings.trendsRange,
   145→   145→  );
   146→   146→  const [metric, setMetric] = useState<MetricId>(settings.trendsMetric);
   147→   147→
   148→   148→  const filtered = useMemo(() => filterEntries(entries, rangeSel), [entries, rangeSel]);
   149→   149→
   150→   150→  /* ── Geometry ── */
   151→   151→  const geo = useMemo(() => {
   152→   152→    if (filtered.length < 2) return null;
   153→   153→
   154→   154→    // Ascending by time
   155→   155→    const sorted = [...filtered].sort((a, b) => a.reading.timestamp - b.reading.timestamp);
   156→   156→    const getV = (e: HistoryEntry) =>
   157→   157→      metric === 'moisture' ? e.reading.moisture : e.reading.temperature;
   158→   158→    // Temperature is required for temp mode — tolerate legacy entries without it
   159→   159→    const usable = metric === 'temp'
   160→   160→      ? sorted.filter((e) => typeof e.reading.temperature === 'number')
   161→   161→      : sorted;
   162→   162→    if (usable.length < 2) return null;
   163→   163→
   164→   164→    const values = usable.map(getV);
   165→   165→
   166→   166→    const tMin = usable[0].reading.timestamp;
   167→   167→    const tMax = usable[usable.length - 1].reading.timestamp;
   168→   168→    const tRange = tMax - tMin || 1;
   169→   169→
   170→   170→    let lo: number;
   171→   171→    let hi: number;
   172→   172→    let vPad: number;
   173→   173→    if (metric === 'moisture') {
   174→   174→      // Include thresholds in the domain so zone bands are always visible
   175→   175→      lo = Math.min(...values, safe);
   176→   176→      hi = Math.max(...values, critical);
   177→   177→      vPad = Math.max((hi - lo) * 0.25, 1.2);
   178→   178→    } else {
   179→   179→      lo = Math.min(...values);
   180→   180→      hi = Math.max(...values);
   181→   181→      vPad = Math.max((hi - lo) * 0.3, 1.5);
   182→   182→    }
   183→   183→    const vLo = lo - vPad;
   184→   184→    const vHi = hi + vPad;
   185→   185→    const vSpan = vHi - vLo;
   186→   186→
   187→   187→    const drawW = W - PAD_X * 2;
   188→   188→    const drawH = H - PAD_TOP - PAD_BOT;
   189→   189→
   190→   190→    const xOf = (ts: number) => PAD_X + ((ts - tMin) / tRange) * drawW;
   191→   191→    const yOf = (v: number) => PAD_TOP + drawH - ((v - vLo) / vSpan) * drawH;
   192→   192→
   193→   193→    const pts: TrendPt[] = usable.map((e) => ({
   194→   194→      x: xOf(e.reading.timestamp),
   195→   195→      y: yOf(getV(e)),
   196→   196→      v: getV(e),
   197→   197→      t: e.reading.timestamp,
   198→   198→      state: e.decision.state,
   199→   199→    }));
   200→   200→
   201→   201→    // Trailing moving average overlay (same x coordinates as the raw curve)
   202→   202→    const maValues = movingAverage(values);
   203→   203→    const maPts = pts.map((p, i) => ({ x: p.x, y: yOf(maValues[i]) }));
   204→   204→    let maPathD = '';
   205→   205→    if (maPts.length >= 2) {
   206→   206→      maPathD = `M${maPts[0].x.toFixed(2)},${maPts[0].y.toFixed(2)}`;
   207→   207→      for (const seg of monotoneCubic(maPts)) {
   208→   208→        maPathD += ` C${seg.cp1x.toFixed(2)},${seg.cp1y.toFixed(2)} ${seg.cp2x.toFixed(2)},${seg.cp2y.toFixed(2)} ${seg.x.toFixed(2)},${seg.y.toFixed(2)}`;
   209→   209→      }
   210→   210→    }
   211→   211→
   212→   212→    // Curve path
   213→   213→    let pathD = `M${pts[0].x.toFixed(2)},${pts[0].y.toFixed(2)}`;
   214→   214→    for (const seg of monotoneCubic(pts)) {
   215→   215→      pathD += ` C${seg.cp1x.toFixed(2)},${seg.cp1y.toFixed(2)} ${seg.cp2x.toFixed(2)},${seg.cp2y.toFixed(2)} ${seg.x.toFixed(2)},${seg.y.toFixed(2)}`;
   216→   216→    }
   217→   217→    const fillD = `${pathD} L${pts[pts.length - 1].x.toFixed(2)},${H - PAD_BOT} L${pts[0].x.toFixed(2)},${H - PAD_BOT} Z`;
   218→   218→
   219→   219→    // Zone bands (clamped to the plot area) — moisture only (thresholds are moisture-specific)
   220→   220→    const plotBottom = H - PAD_BOT;
   221→   221→    const band = (vFrom: number, vTo: number) => {
   222→   222→      const yA = Math.max(PAD_TOP, Math.min(plotBottom, yOf(vFrom)));
   223→   223→      const yB = Math.max(PAD_TOP, Math.min(plotBottom, yOf(vTo)));
   224→   224→      return { y: Math.min(yA, yB), h: Math.abs(yB - yA) };
   225→   225→    };
   226→   226→    const critBand = band(Math.max(critical, vLo), vHi);        // above critical
   227→   227→    const warnBand = band(safe, critical);                       // between safe and critical
   228→   228→    const safeBand = band(vLo, safe);                            // below safe
   229→   229→
   230→   230→    // Stats
   231→   231→    const avg = values.reduce((s, v) => s + v, 0) / values.length;
   232→   232→    const min = Math.min(...values);
   233→   233→    const max = Math.max(...values);
   234→   234→
   235→   235→    // Dot thinning for long ranges
   236→   236→    const dotStep = pts.length > 60 ? Math.ceil(pts.length / 48) : 1;
   237→   237→
   238→   238→    // Time axis labels: first / mid / last
   239→   239→    const midIdx = Math.floor(pts.length / 2);
   240→   240→    const fmt = (ts: number) => (rangeSel === '24h' ? format(ts, 'HH:mm') : format(ts, 'MMM d'));
   241→   241→    const timeLabels = [
   242→   242→      { x: pts[0].x, anchor: 'start' as const, label: fmt(tMin) },
   243→   243→      ...(pts.length > 6 ? [{ x: pts[midIdx].x, anchor: 'middle' as const, label: fmt(usable[midIdx].reading.timestamp) }] : []),
   244→   244→      { x: pts[pts.length - 1].x, anchor: 'end' as const, label: fmt(tMax) },
   245→   245→    ];
   246→   246→
   247→   247→    return {
   248→   248→      pts, pathD, fillD, critBand, warnBand, safeBand, avg, min, max,
   249→   249→      avgY: yOf(avg), dotStep, timeLabels,
   250→   250→      safeY: yOf(safe), criticalY: yOf(critical),
   251→   251→      trend: trendDirection(values[values.length - 1] - values[0], metric),
   252→   252→      maPathD, maValues,
   253→   253→    };
   254→   254→  }, [filtered, safe, critical, rangeSel, metric]);
   255→   255→
   256→   256→  /* ── Scrub state (timestamp-anchored, same pattern as hero sparkline) ── */
   257→   257→  const [scrubSel, setScrubSel] = useState<{ t: number; leftPx: number; topPx: number } | null>(null);
   258→   258→  const svgRef = useRef<SVGSVGElement | null>(null);
   259→   259→  const wrapRef = useRef<HTMLDivElement | null>(null);
   260→   260→  const draggingRef = useRef(false);
   261→   261→  const autoClearRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   262→   262→
   263→   263→  const scrubPt = useMemo(
   264→   264→    () => (scrubSel && geo ? geo.pts.find((p) => p.t === scrubSel.t) ?? null : null),
   265→   265→    [geo, scrubSel],
   266→   266→  );
   267→   267→  const scrubIdx = useMemo(
   268→   268→    () => (geo && scrubSel ? geo.pts.findIndex((p) => p.t === scrubSel.t) : -1),
   269→   269→    [geo, scrubSel],
   270→   270→  );
   271→   271→
   272→   272→  const clearAutoTimer = useCallback(() => {
   273→   273→    if (autoClearRef.current) {
   274→   274→      clearTimeout(autoClearRef.current);
   275→   275→      autoClearRef.current = null;
   276→   276→    }
   277→   277→  }, []);
   278→   278→
   279→   279→  const clearScrub = useCallback(() => {
   280→   280→    clearAutoTimer();
   281→   281→    draggingRef.current = false;
   282→   282→    setScrubSel(null);
   283→   283→  }, [clearAutoTimer]);
   284→   284→
   285→   285→  useEffect(() => () => clearAutoTimer(), [clearAutoTimer]);
   286→   286→
   287→   287→  /** Switch range or metric: clears any active scrub (selection is scope-scoped)
   288→   288→   *  and persists the choice so the chart reopens in the same configuration. */
   289→   289→  const selectRange = useCallback(
   290→   290→    (id: RangeId) => {
   291→   291→      clearAutoTimer();
   292→   292→      draggingRef.current = false;
   293→   293→      setScrubSel(null);
   294→   294→      setRangeSel(id);
   295→   295→      updateSettings({ trendsRange: id });
   296→   296→    },
   297→   297→    [clearAutoTimer, updateSettings],
   298→   298→  );
   299→   299→
   300→   300→  const selectMetric = useCallback(
   301→   301→    (m: MetricId) => {
   302→   302→      clearAutoTimer();
   303→   303→      draggingRef.current = false;
   304→   304→      setScrubSel(null);
   305→   305→      setMetric(m);
   306→   306→      updateSettings({ trendsMetric: m });
   307→   307→    },
   308→   308→    [clearAutoTimer, updateSettings],
   309→   309→  );
   310→   310→
   311→   311→  const anchorFor = useCallback(
   312→   312→    (i: number): { leftPx: number; topPx: number } | null => {
   313→   313→      const svg = svgRef.current;
   314→   314→      const wrap = wrapRef.current;
   315→   315→      if (!svg || !wrap || !geo || i < 0 || i >= geo.pts.length) return null;
   316→   316→      const ctm = svg.getScreenCTM();
   317→   317→      if (!ctm) return null;
   318→   318→      const pt = geo.pts[i];
   319→   319→      let cx: number;
   320→   320→      let cy: number;
   321→   321→      if (typeof DOMPoint === 'function') {
   322→   322→        const sp = new DOMPoint(pt.x, pt.y).matrixTransform(ctm);
   323→   323→        cx = sp.x; cy = sp.y;
   324→   324→      } else {
   325→   325→        cx = pt.x * ctm.a + pt.y * ctm.c + ctm.e;
   326→   326→        cy = pt.x * ctm.b + pt.y * ctm.d + ctm.f;
   327→   327→      }
   328→   328→      const rect = wrap.getBoundingClientRect();
   329→   329→      const leftPx = Math.min(Math.max(cx - rect.left, TOOLTIP_CLAMP), Math.max(rect.width - TOOLTIP_CLAMP, TOOLTIP_CLAMP));
   330→   330→      const topPx = cy - rect.top;
   331→   331→      return { leftPx, topPx };
   332→   332→    },
   333→   333→    [geo],
   334→   334→  );
   335→   335→
   336→   336→  const nearestIndex = useCallback(
   337→   337→    (vx: number): number => {
   338→   338→      if (!geo || geo.pts.length === 0) return -1;
   339→   339→      const pts = geo.pts;
   340→   340→      let lo = 0;
   341→   341→      let hi = pts.length - 1;
   342→   342→      while (hi - lo > 1) {
   343→   343→        const mid = (lo + hi) >> 1;
   344→   344→        if (pts[mid].x < vx) lo = mid;
   345→   345→        else hi = mid;
   346→   346→      }
   347→   347→      return Math.abs(pts[lo].x - vx) <= Math.abs(pts[hi].x - vx) ? lo : hi;
   348→   348→    },
   349→   349→    [geo],
   350→   350→  );
   351→   351→
   352→   352→  const scrubTo = useCallback(
   353→   353→    (i: number) => {
   354→   354→      const anchor = anchorFor(i);
   355→   355→      if (!anchor || !geo) return;
   356→   356→      const sel = { t: geo.pts[i].t, ...anchor };
   357→   357→      setScrubSel((prev) => {
   358→   358→        if (prev && prev.t !== sel.t) haptic(settings.hapticsEnabled, 4);
   359→   359→        return sel;
   360→   360→      });
   361→   361→    },
   362→   362→    [anchorFor, geo, settings.hapticsEnabled],
   363→   363→  );
   364→   364→
   365→   365→  const clientToViewBox = useCallback((clientX: number, clientY: number) => {
   366→   366→    const svg = svgRef.current;
   367→   367→    if (!svg) return null;
   368→   368→    const ctm = svg.getScreenCTM();
   369→   369→    if (!ctm) return null;
   370→   370→    if (typeof DOMPoint === 'function') {
   371→   371→      const p = new DOMPoint(clientX, clientY).matrixTransform(ctm.inverse());
   372→   372→      return { x: p.x, y: p.y };
   373→   373→    }
   374→   374→    const inv = ctm.inverse();
   375→   375→    return {
   376→   376→      x: clientX * inv.a + clientY * inv.c + inv.e,
   377→   377→      y: clientX * inv.b + clientY * inv.d + inv.f,
   378→   378→    };
   379→   379→  }, []);
   380→   380→
   381→   381→  const onPointerDown = useCallback(
   382→   382→    (e: React.PointerEvent<SVGSVGElement>) => {
   383→   383→      if (!geo || geo.pts.length < 2) return;
   384→   384→      clearAutoTimer();
   385→   385→      draggingRef.current = true;
   386→   386→      try { svgRef.current?.setPointerCapture(e.pointerId); } catch { /* no-op */ }
   387→   387→      const p = clientToViewBox(e.clientX, e.clientY);
   388→   388→      if (p) scrubTo(nearestIndex(p.x));
   389→   389→    },
   390→   390→    [geo, clearAutoTimer, clientToViewBox, nearestIndex, scrubTo],
   391→   391→  );
   392→   392→
   393→   393→  const onPointerMove = useCallback(
   394→   394→    (e: React.PointerEvent<SVGSVGElement>) => {
   395→   395→      if (!geo || geo.pts.length < 2) return;
   396→   396→      const hoverScrub = e.pointerType === 'mouse' && e.buttons === 0;
   397→   397→      if (!draggingRef.current && !hoverScrub) return;
   398→   398→      const p = clientToViewBox(e.clientX, e.clientY);
   399→   399→      if (p) scrubTo(nearestIndex(p.x));
   400→   400→    },
   401→   401→    [geo, clientToViewBox, nearestIndex, scrubTo],
   402→   402→  );
   403→   403→
   404→   404→  const onPointerUp = useCallback(
   405→   405→    (e: React.PointerEvent<SVGSVGElement>) => {
   406→   406→      draggingRef.current = false;
   407→   407→      try { svgRef.current?.releasePointerCapture(e.pointerId); } catch { /* no-op */ }
   408→   408→      if (e.pointerType !== 'mouse') {
