    23→    23→type RangeId = Exclude<TrendsRange, 'auto'>;
    24→    24→type MetricId = TrendsMetric;
    25→    25→
    26→    26→const RANGES: { id: RangeId; label: string; ms: number | null }[] = [
    27→    27→  { id: '24h', label: '24H', ms: 24 * 3600e3 },
    28→    28→  { id: '7d', label: '7D', ms: 7 * 24 * 3600e3 },
    29→    29→  { id: '30d', label: '30D', ms: 30 * 24 * 3600e3 },
    30→    30→  { id: 'all', label: 'ALL', ms: null },
    31→    31→];
    32→    32→
    33→    33→const SCRUB_AUTO_CLEAR_MS = 3200;
    34→    34→const TOOLTIP_CLAMP = 50;
    35→    35→
    36→    36→const W = 358;
    37→    37→const H = 216;
    38→    38→const PAD_X = 18;
    39→    39→const PAD_TOP = 18;
    40→    40→const PAD_BOT = 24;
    41→    41→
    42→    42→interface TrendPt {
    43→    43→  x: number;
    44→    44→  y: number;
    45→    45→  v: number;
    46→    46→  t: number;
    47→    47→  state: string;

    50→    50→interface TrendDir {
    51→    51→  dir: 'up' | 'down' | 'flat';
    52→    52→  delta: number;
    53→    53→  labelKey: string;
    54→    54→}
    55→    55→
    56→    56→/** Direction + signed delta across the visible range (per metric) */
    57→    57→function trendDirection(delta: number, metric: MetricId): TrendDir {
    58→    58→  const eps = metric === 'moisture' ? 0.15 : 0.3;
    59→    59→  if (delta > eps) return { dir: 'up', delta, labelKey: 'history.trend_rising' };
    60→    60→  if (delta < -eps) return { dir: 'down', delta, labelKey: 'history.trend_falling' };
    61→    61→  return { dir: 'flat', delta, labelKey: 'history.trend_stable' };
    62→    62→}
    63→    63→
    64→    64→function filterEntries(entries: HistoryEntry[], range: RangeId): HistoryEntry[] {
    65→    65→  const r = RANGES.find((x) => x.id === range);
    66→    66→  if (!r || r.ms == null) return entries;
    67→    67→  const cutoff = Date.now() - r.ms;
    68→    68→  return entries.filter((e) => e.reading.timestamp >= cutoff);
    69→    69→}
    70→    70→
    71→    71→/** Smallest range with ≥2 points, falling back to ALL */
    72→    72→function detectRange(entries: HistoryEntry[]): RangeId {
    73→    73→  for (const r of RANGES) {
    74→    74→    if (filterEntries(entries, r.id).length >= 2) return r.id;
    75→    75→  }
    76→    76→  return 'all';
    77→    77→}
    78→    78→
    79→    79→/** Monotone cubic Bézier (Fritsch–Carlson) — shared curve algorithm */
    80→    80→function monotoneCubic(pts: { x: number; y: number }[]) {
    81→    81→  const n = pts.length;
    82→    82→  if (n < 2) return [];
    83→    83→  const dx: number[] = [];
    84→    84→  const dy: number[] = [];
    85→    85→  const m: number[] = [];
    86→    86→  for (let i = 0; i < n - 1; i++) {
    87→    87→    dx.push(pts[i + 1].x - pts[i].x);
    88→    88→    dy.push(pts[i + 1].y - pts[i].y);
    89→    89→    m.push(dy[i] / (dx[i] || 1));
    90→    90→  }
    91→    91→  const c: number[] = [m[0]];
    92→    92→  for (let i = 1; i < n - 1; i++) {
    93→    93→    if (m[i - 1] * m[i] <= 0) c.push(0);
    94→    94→    else {
    95→    95→      const common = dx[i - 1] + dx[i];
    96→    96→      c.push((3 * common) / ((common + dx[i]) / m[i - 1] + (common + dx[i - 1]) / m[i]));
    97→    97→    }
    98→    98→  }
    99→    99→  c.push(m[n - 2]);
   100→   100→  for (let i = 0; i < n - 1; i++) {
   101→   101→    const acc = c[i] / (m[i] || 1);
   102→   102→    const succ = c[i + 1] / (m[i] || 1);
   103→   103→    if (acc < 0 || succ < 0) {
   104→   104→      c[i] = 0;
   105→   105→      c[i + 1] = 0;
   106→   106→    }
   107→   107→  }
   108→   108→  const segs: { x: number; y: number; cp1x: number; cp1y: number; cp2x: number; cp2y: number }[] = [];
   109→   109→  for (let i = 0; i < n - 1; i++) {
   110→   110→    const p0 = pts[i];
   111→   111→    const p1 = pts[i + 1];
   112→   112→    const segDx = dx[i];
   113→   113→    segs.push({
   114→   114→      x: p1.x, y: p1.y,
   115→   115→      cp1x: p0.x + segDx / 3, cp1y: p0.y + (segDx / 3) * c[i],
   116→   116→      cp2x: p1.x - segDx / 3, cp2y: p1.y - (segDx / 3) * c[i + 1],
   117→   117→    });
   118→   118→  }
   119→   119→  return segs;
   120→   120→}
   121→   121→
   122→   122→export function HistoryTrends({ entries }: { entries: HistoryEntry[] }) {
   123→   123→  const { settings, updateSettings } = useGrainStore();
   124→   124→  const lang = settings.language as AppLanguage;
   125→   125→  const accentHex = getAccentHex(settings.accentColor);
   126→   126→  const { safe, warn, critical } = settings.thresholds;
   127→   127→
   128→   128→  // Metric + range persist across sessions via AppSettings ('auto' re-detects)
   129→   129→  const [rangeSel, setRangeSel] = useState<RangeId>(() =>
   130→   130→    settings.trendsRange === 'auto' ? detectRange(entries) : settings.trendsRange,
   131→   131→  );
   132→   132→  const [metric, setMetric] = useState<MetricId>(settings.trendsMetric);
   133→   133→
   134→   134→  const filtered = useMemo(() => filterEntries(entries, rangeSel), [entries, rangeSel]);
   135→   135→
   136→   136→  /* ── Geometry ── */
   137→   137→  const geo = useMemo(() => {
   138→   138→    if (filtered.length < 2) return null;
   139→   139→
   140→   140→    // Ascending by time
   141→   141→    const sorted = [...filtered].sort((a, b) => a.reading.timestamp - b.reading.timestamp);
   142→   142→    const getV = (e: HistoryEntry) =>
   143→   143→      metric === 'moisture' ? e.reading.moisture : e.reading.temperature;
   144→   144→    // Temperature is required for temp mode — tolerate legacy entries without it
   145→   145→    const usable = metric === 'temp'
   146→   146→      ? sorted.filter((e) => typeof e.reading.temperature === 'number')
   147→   147→      : sorted;
   148→   148→    if (usable.length < 2) return null;
   149→   149→
   150→   150→    const values = usable.map(getV);
   151→   151→
   152→   152→    const tMin = usable[0].reading.timestamp;
   153→   153→    const tMax = usable[usable.length - 1].reading.timestamp;
   154→   154→    const tRange = tMax - tMin || 1;
   155→   155→
   156→   156→    let lo: number;
   157→   157→    let hi: number;
   158→   158→    let vPad: number;
   159→   159→    if (metric === 'moisture') {
   160→   160→      // Include thresholds in the domain so zone bands are always visible
   161→   161→      lo = Math.min(...values, safe);
   162→   162→      hi = Math.max(...values, critical);
   163→   163→      vPad = Math.max((hi - lo) * 0.25, 1.2);
   164→   164→    } else {
   165→   165→      lo = Math.min(...values);
   166→   166→      hi = Math.max(...values);
   167→   167→      vPad = Math.max((hi - lo) * 0.3, 1.5);
   168→   168→    }
   169→   169→    const vLo = lo - vPad;
   170→   170→    const vHi = hi + vPad;
   171→   171→    const vSpan = vHi - vLo;
   172→   172→
   173→   173→    const drawW = W - PAD_X * 2;
   174→   174→    const drawH = H - PAD_TOP - PAD_BOT;
   175→   175→
   176→   176→    const xOf = (ts: number) => PAD_X + ((ts - tMin) / tRange) * drawW;
   177→   177→    const yOf = (v: number) => PAD_TOP + drawH - ((v - vLo) / vSpan) * drawH;
   178→   178→
   179→   179→    const pts: TrendPt[] = usable.map((e) => ({
   180→   180→      x: xOf(e.reading.timestamp),
   181→   181→      y: yOf(getV(e)),
   182→   182→      v: getV(e),
   183→   183→      t: e.reading.timestamp,
   184→   184→      state: e.decision.state,
   185→   185→    }));
   186→   186→
   187→   187→    // Curve path
   188→   188→    let pathD = `M${pts[0].x.toFixed(2)},${pts[0].y.toFixed(2)}`;
   189→   189→    for (const seg of monotoneCubic(pts)) {
   190→   190→      pathD += ` C${seg.cp1x.toFixed(2)},${seg.cp1y.toFixed(2)} ${seg.cp2x.toFixed(2)},${seg.cp2y.toFixed(2)} ${seg.x.toFixed(2)},${seg.y.toFixed(2)}`;
   191→   191→    }
   192→   192→    const fillD = `${pathD} L${pts[pts.length - 1].x.toFixed(2)},${H - PAD_BOT} L${pts[0].x.toFixed(2)},${H - PAD_BOT} Z`;
   193→   193→
   194→   194→    // Zone bands (clamped to the plot area) — moisture only (thresholds are moisture-specific)
   195→   195→    const plotBottom = H - PAD_BOT;
   196→   196→    const band = (vFrom: number, vTo: number) => {
   197→   197→      const yA = Math.max(PAD_TOP, Math.min(plotBottom, yOf(vFrom)));
   198→   198→      const yB = Math.max(PAD_TOP, Math.min(plotBottom, yOf(vTo)));
   199→   199→      return { y: Math.min(yA, yB), h: Math.abs(yB - yA) };
   200→   200→    };
   201→   201→    const critBand = band(Math.max(critical, vLo), vHi);        // above critical
   202→   202→    const warnBand = band(safe, critical);                       // between safe and critical
   203→   203→    const safeBand = band(vLo, safe);                            // below safe
   204→   204→
   205→   205→    // Stats
   206→   206→    const avg = values.reduce((s, v) => s + v, 0) / values.length;
   207→   207→    const min = Math.min(...values);
   208→   208→    const max = Math.max(...values);
   209→   209→
   210→   210→    // Dot thinning for long ranges
   211→   211→    const dotStep = pts.length > 60 ? Math.ceil(pts.length / 48) : 1;
   212→   212→
   213→   213→    // Time axis labels: first / mid / last
   214→   214→    const midIdx = Math.floor(pts.length / 2);
   215→   215→    const fmt = (ts: number) => (rangeSel === '24h' ? format(ts, 'HH:mm') : format(ts, 'MMM d'));
   216→   216→    const timeLabels = [
   217→   217→      { x: pts[0].x, anchor: 'start' as const, label: fmt(tMin) },
   218→   218→      ...(pts.length > 6 ? [{ x: pts[midIdx].x, anchor: 'middle' as const, label: fmt(usable[midIdx].reading.timestamp) }] : []),
   219→   219→      { x: pts[pts.length - 1].x, anchor: 'end' as const, label: fmt(tMax) },
   220→   220→    ];
   221→   221→
   222→   222→    return {
   223→   223→      pts, pathD, fillD, critBand, warnBand, safeBand, avg, min, max,
   224→   224→      avgY: yOf(avg), dotStep, timeLabels,
   225→   225→      safeY: yOf(safe), criticalY: yOf(critical),
   226→   226→      trend: trendDirection(values[values.length - 1] - values[0], metric),
   227→   227→    };
   228→   228→  }, [filtered, safe, critical, rangeSel, metric]);
   229→   229→
   230→   230→  /* ── Scrub state (timestamp-anchored, same pattern as hero sparkline) ── */
   231→   231→  const [scrubSel, setScrubSel] = useState<{ t: number; leftPx: number; topPx: number } | null>(null);
   232→   232→  const svgRef = useRef<SVGSVGElement | null>(null);
   233→   233→  const wrapRef = useRef<HTMLDivElement | null>(null);
   234→   234→  const draggingRef = useRef(false);
   235→   235→  const autoClearRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   236→   236→
   237→   237→  const scrubPt = useMemo(
   238→   238→    () => (scrubSel && geo ? geo.pts.find((p) => p.t === scrubSel.t) ?? null : null),
   239→   239→    [geo, scrubSel],
   240→   240→  );
   241→   241→  const scrubIdx = useMemo(
   242→   242→    () => (geo && scrubSel ? geo.pts.findIndex((p) => p.t === scrubSel.t) : -1),
   243→   243→    [geo, scrubSel],
   244→   244→  );
   245→   245→
   246→   246→  const clearAutoTimer = useCallback(() => {
   247→   247→    if (autoClearRef.current) {
   248→   248→      clearTimeout(autoClearRef.current);
   249→   249→      autoClearRef.current = null;
   250→   250→    }
   251→   251→  }, []);
   252→   252→
   253→   253→  const clearScrub = useCallback(() => {
   254→   254→    clearAutoTimer();
   255→   255→    draggingRef.current = false;
   256→   256→    setScrubSel(null);
   257→   257→  }, [clearAutoTimer]);
   258→   258→
   259→   259→  useEffect(() => () => clearAutoTimer(), [clearAutoTimer]);
   260→   260→
   261→   261→  /** Switch range or metric: clears any active scrub (selection is scope-scoped)
   262→   262→   *  and persists the choice so the chart reopens in the same configuration. */
   263→   263→  const selectRange = useCallback(
   264→   264→    (id: RangeId) => {
   265→   265→      clearAutoTimer();
   266→   266→      draggingRef.current = false;
   267→   267→      setScrubSel(null);
   268→   268→      setRangeSel(id);
   269→   269→      updateSettings({ trendsRange: id });
   270→   270→    },
   271→   271→    [clearAutoTimer, updateSettings],
   272→   272→  );
   273→   273→
   274→   274→  const selectMetric = useCallback(
   275→   275→    (m: MetricId) => {
   276→   276→      clearAutoTimer();
   277→   277→      draggingRef.current = false;
   278→   278→      setScrubSel(null);
   279→   279→      setMetric(m);
   280→   280→      updateSettings({ trendsMetric: m });
   281→   281→    },
   282→   282→    [clearAutoTimer, updateSettings],
   283→   283→  );
   284→   284→
   285→   285→  const anchorFor = useCallback(
   286→   286→    (i: number): { leftPx: number; topPx: number } | null => {
   287→   287→      const svg = svgRef.current;
   288→   288→      const wrap = wrapRef.current;
   289→   289→      if (!svg || !wrap || !geo || i < 0 || i >= geo.pts.length) return null;
   290→   290→      const ctm = svg.getScreenCTM();
   291→   291→      if (!ctm) return null;
   292→   292→      const pt = geo.pts[i];
   293→   293→      let cx: number;
   294→   294→      let cy: number;
   295→   295→      if (typeof DOMPoint === 'function') {
   296→   296→        const sp = new DOMPoint(pt.x, pt.y).matrixTransform(ctm);
   297→   297→        cx = sp.x; cy = sp.y;
   298→   298→      } else {
   299→   299→        cx = pt.x * ctm.a + pt.y * ctm.c + ctm.e;
   300→   300→        cy = pt.x * ctm.b + pt.y * ctm.d + ctm.f;
   301→   301→      }
   302→   302→      const rect = wrap.getBoundingClientRect();
   303→   303→      const leftPx = Math.min(Math.max(cx - rect.left, TOOLTIP_CLAMP), Math.max(rect.width - TOOLTIP_CLAMP, TOOLTIP_CLAMP));
   304→   304→      const topPx = cy - rect.top;
   305→   305→      return { leftPx, topPx };
   306→   306→    },
   307→   307→    [geo],
   308→   308→  );
   309→   309→
   310→   310→  const nearestIndex = useCallback(
   311→   311→    (vx: number): number => {
   312→   312→      if (!geo || geo.pts.length === 0) return -1;
   313→   313→      const pts = geo.pts;
   314→   314→      let lo = 0;
   315→   315→      let hi = pts.length - 1;
   316→   316→      while (hi - lo > 1) {
   317→   317→        const mid = (lo + hi) >> 1;
   318→   318→        if (pts[mid].x < vx) lo = mid;
   319→   319→        else hi = mid;
   320→   320→      }
   321→   321→      return Math.abs(pts[lo].x - vx) <= Math.abs(pts[hi].x - vx) ? lo : hi;
   322→   322→    },
   323→   323→    [geo],
   324→   324→  );
   325→   325→
   326→   326→  const scrubTo = useCallback(
   327→   327→    (i: number) => {
   328→   328→      const anchor = anchorFor(i);
   329→   329→      if (!anchor || !geo) return;
   330→   330→      const sel = { t: geo.pts[i].t, ...anchor };
   331→   331→      setScrubSel((prev) => {
   332→   332→        if (prev && prev.t !== sel.t) haptic(settings.hapticsEnabled, 4);
   333→   333→        return sel;
   334→   334→      });
   335→   335→    },
   336→   336→    [anchorFor, geo, settings.hapticsEnabled],
   337→   337→  );
   338→   338→
   339→   339→  const clientToViewBox = useCallback((clientX: number, clientY: number) => {
   340→   340→    const svg = svgRef.current;
   341→   341→    if (!svg) return null;
   342→   342→    const ctm = svg.getScreenCTM();
   343→   343→    if (!ctm) return null;
   344→   344→    if (typeof DOMPoint === 'function') {
   345→   345→      const p = new DOMPoint(clientX, clientY).matrixTransform(ctm.inverse());
   346→   346→      return { x: p.x, y: p.y };
   347→   347→    }
   348→   348→    const inv = ctm.inverse();
   349→   349→    return {
   350→   350→      x: clientX * inv.a + clientY * inv.c + inv.e,
   351→   351→      y: clientX * inv.b + clientY * inv.d + inv.f,
   352→   352→    };
   353→   353→  }, []);
   354→   354→
   355→   355→  const onPointerDown = useCallback(
   356→   356→    (e: React.PointerEvent<SVGSVGElement>) => {
   357→   357→      if (!geo || geo.pts.length < 2) return;
   358→   358→      clearAutoTimer();
   359→   359→      draggingRef.current = true;
   360→   360→      try { svgRef.current?.setPointerCapture(e.pointerId); } catch { /* no-op */ }
   361→   361→      const p = clientToViewBox(e.clientX, e.clientY);
   362→   362→      if (p) scrubTo(nearestIndex(p.x));
   363→   363→    },
   364→   364→    [geo, clearAutoTimer, clientToViewBox, nearestIndex, scrubTo],
   365→   365→  );
   366→   366→
   367→   367→  const onPointerMove = useCallback(
   368→   368→    (e: React.PointerEvent<SVGSVGElement>) => {
   369→   369→      if (!geo || geo.pts.length < 2) return;
   370→   370→      const hoverScrub = e.pointerType === 'mouse' && e.buttons === 0;
   371→   371→      if (!draggingRef.current && !hoverScrub) return;
   372→   372→      const p = clientToViewBox(e.clientX, e.clientY);
   373→   373→      if (p) scrubTo(nearestIndex(p.x));
   374→   374→    },
   375→   375→    [geo, clientToViewBox, nearestIndex, scrubTo],
   376→   376→  );
   377→   377→
   378→   378→  const onPointerUp = useCallback(
   379→   379→    (e: React.PointerEvent<SVGSVGElement>) => {
   380→   380→      draggingRef.current = false;
   381→   381→      try { svgRef.current?.releasePointerCapture(e.pointerId); } catch { /* no-op */ }
   382→   382→      if (e.pointerType !== 'mouse') {
   383→   383→        clearAutoTimer();
   384→   384→        autoClearRef.current = setTimeout(() => setScrubSel(null), SCRUB_AUTO_CLEAR_MS);
   385→   385→      }
   386→   386→    },
   387→   387→    [clearAutoTimer],
   388→   388→  );
   389→   389→
   390→   390→  const onPointerLeave = useCallback(
   391→   391→    (e: React.PointerEvent<SVGSVGElement>) => {
   392→   392→      if (e.pointerType === 'mouse' && !draggingRef.current) setScrubSel(null);
   393→   393→    },
   394→   394→    [],
   395→   395→  );
   396→   396→
   397→   397→  const onKeyDown = useCallback(
   398→   398→    (e: React.KeyboardEvent<HTMLDivElement>) => {
   399→   399→      if (!geo || geo.pts.length < 2) return;
   400→   400→      const lastIdx = geo.pts.length - 1;
   401→   401→      if (e.key === 'Escape') { clearScrub(); return; }
   402→   402→      let next: number | null = null;
   403→   403→      const cur = scrubIdx >= 0 ? scrubIdx : null;
   404→   404→      if (e.key === 'ArrowLeft') next = Math.max(0, (cur ?? lastIdx) - 1);
   405→   405→      else if (e.key === 'ArrowRight') next = Math.min(lastIdx, (cur ?? lastIdx - 1) + 1);
   406→   406→      else if (e.key === 'Home') next = 0;
   407→   407→      else if (e.key === 'End') next = lastIdx;
   408→   408→      if (next === null) return;
   409→   409→      e.preventDefault();
   410→   410→      clearAutoTimer();
   411→   411→      scrubTo(next);
   412→   412→    },
   413→   413→    [geo, scrubIdx, clearScrub, clearAutoTimer, scrubTo],
   414→   414→  );
   415→   415→
