   339→function AdvancedSettingsContent() {
   340→  const {
   341→    settings, updateSettings,
   342→    disconnectProbe, sendProbeCommand,
   343→    syncProbeHistory, clearHistory,
   344→    hasDevice, showToast,
   345→    demoMode, enableDemoMode, disableDemoMode,
   346→    demoIntervalSec, setDemoIntervalSec,
   347→    loadHistory, loadSettings,
   348→  } = useGrainStore();
   349→
   350→  const lang = settings.language as AppLanguage;
   351→  const hasConnection = hasDevice && !demoMode;
   352→  const fileInputRef = useRef<HTMLInputElement | null>(null);
   353→  const importModeRef = useRef<'merge' | 'replace'>('merge');
   354→  const [confirmReplace, setConfirmReplace] = useState(false);
   355→  const overlayRef = useRef<HTMLDivElement | null>(null);
   356→  const lastFocusRef = useRef<HTMLElement | null>(null);
   357→
   358→  /* Focus trap + focus restore for the destructive confirmation modal.
   359→     Tab/Shift+Tab cycle inside the dialog; Escape closes (capture-phase,
   360→     works even when focus sits on body); on close, focus returns to the
   361→     element that opened the dialog (captured in the click handler — by the
   362→     time the open-effect runs, React has already applied autoFocus to the
   363→     Cancel button, so the effect would otherwise capture the dialog itself). */
   364→  const openReplaceModal = () => {
   365→    lastFocusRef.current = document.activeElement as HTMLElement | null;
   366→    setConfirmReplace(true);
   367→  };
   368→
   369→  useEffect(() => {
   370→    if (!confirmReplace) return;
   371→    const onKey = (e: KeyboardEvent) => {
   372→      if (e.key === 'Escape') {
   373→        e.stopPropagation();
   374→        setConfirmReplace(false);
   375→        return;
   376→      }
   377→      if (e.key !== 'Tab') return;
   378→      const overlay = overlayRef.current;
   379→      if (!overlay) return;
   380→      const focusables = Array.from(
   381→        overlay.querySelectorAll<HTMLElement>(
   382→          'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])',
   383→        ),
   384→      ).filter((el) => !el.hasAttribute('disabled'));
   385→      if (focusables.length === 0) return;
   386→      const first = focusables[0];
   387→      const last = focusables[focusables.length - 1];
   388→      const active = document.activeElement;
   389→      if (e.shiftKey && (active === first || !overlay.contains(active))) {
   390→        e.preventDefault();
   391→        last.focus();
   392→      } else if (!e.shiftKey && (active === last || !overlay.contains(active))) {
   393→        e.preventDefault();
   394→        first.focus();
   395→      }
   396→    };
   397→    document.addEventListener('keydown', onKey, true);
   398→    return () => {
   399→      document.removeEventListener('keydown', onKey, true);
   400→      lastFocusRef.current?.focus?.();
   401→    };
   402→  }, [confirmReplace]);
   403→
   404→  const openImportPicker = (mode: 'merge' | 'replace') => {
   405→    importModeRef.current = mode;
   406→    fileInputRef.current?.click();
   407→  };
   408→
   409→  const handleImportFile = async (e: React.ChangeEvent<HTMLInputElement>) => {
   410→    const file = e.target.files?.[0];
   411→    e.target.value = ''; // allow re-selecting the same file
   412→    if (!file) return;
   413→    const mode = importModeRef.current;
   414→    try {
   415→      const text = await file.text();
   416→      const result = await importBackupJSON(text, mode);
   417→      // Refresh the live store from the restored database
   418→      await loadHistory();
   419→      await loadSettings();
   420→      showToast(`${t(mode === 'replace' ? 'toast.backup_replaced' : 'toast.backup_imported', lang)} · ${result.history}`);
   421→    } catch (err) {
   422→      showToast(err instanceof Error && err.message === 'invalid'
   423→        ? t('toast.import_invalid', lang)
   424→        : t('toast.import_failed', lang));
   425→    }
   426→  };
   427→
   428→  return (
